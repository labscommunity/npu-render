cmake_minimum_required(VERSION 3.25)
project(npu_render_paging_probe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The paging probe requires x64 Windows and MSVC")
endif()
add_executable(nr_paging_probe main.cpp)
target_compile_features(nr_paging_probe PRIVATE cxx_std_17)
target_compile_options(nr_paging_probe PRIVATE /W4 /WX)
target_link_libraries(nr_paging_probe PRIVATE gdi32)
set_property(TARGET nr_paging_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_test(NAME paging_probe_enumeration COMMAND nr_paging_probe --list)
add_library(nr_paging_guard SHARED lab_guard.cpp)
target_compile_features(nr_paging_guard PRIVATE cxx_std_17)
target_compile_options(nr_paging_guard PRIVATE /W4 /WX)
set_property(TARGET nr_paging_guard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(nr_paging_guard_test guard_test.cpp)
target_compile_features(nr_paging_guard_test PRIVATE cxx_std_17)
target_compile_options(nr_paging_guard_test PRIVATE /W4 /WX)
set_property(TARGET nr_paging_guard_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME paging_guard_rejection COMMAND nr_paging_guard_test $<TARGET_FILE:nr_paging_guard>)
