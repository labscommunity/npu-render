cmake_minimum_required(VERSION 3.25)
project(npu_frame_present_probe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The probe requires x64 Windows/MSVC")
endif()
add_executable(nr_frame_present_probe frame_present_probe.cpp)
target_compile_features(nr_frame_present_probe PRIVATE cxx_std_17)
target_compile_definitions(nr_frame_present_probe PRIVATE UNICODE _UNICODE)
target_compile_options(nr_frame_present_probe PRIVATE /W4 /WX)
target_link_libraries(nr_frame_present_probe PRIVATE d3d11 dxgi user32 gdi32 wtsapi32 advapi32)
set_property(TARGET nr_frame_present_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(nr_frame_wait_test frame_wait_test.cpp)
target_compile_features(nr_frame_wait_test PRIVATE cxx_std_17)
target_compile_definitions(nr_frame_wait_test PRIVATE UNICODE _UNICODE)
target_compile_options(nr_frame_wait_test PRIVATE /W4 /WX)
target_link_libraries(nr_frame_wait_test PRIVATE user32)
set_property(TARGET nr_frame_wait_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
