cmake_minimum_required(VERSION 3.25)
project(nr_auto_interactive LANGUAGES CXX)
add_executable(nr_auto_interactive_broker broker.cpp)
add_executable(nr_auto_interactive_consumer consumer.cpp)
add_executable(nr_auto_interactive_contract contract.cpp)
foreach(t nr_auto_interactive_broker nr_auto_interactive_consumer nr_auto_interactive_contract)
 target_compile_features(${t} PRIVATE cxx_std_17)
 target_compile_options(${t} PRIVATE /W4 /WX)
 target_link_libraries(${t} PRIVATE advapi32 bcrypt wtsapi32)
 set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()

enable_testing()
add_test(NAME interactive_lease_contract COMMAND nr_auto_interactive_contract)
