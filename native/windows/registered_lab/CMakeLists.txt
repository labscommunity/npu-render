cmake_minimum_required(VERSION 3.25)
project(npu_render_registered_lab LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The registered lab requires x64 Windows and MSVC")
endif()
set(NR_WDK_INCLUDE_ROOT "" CACHE PATH "WDK Include/version directory")
if(NOT EXISTS "${NR_WDK_INCLUDE_ROOT}/um/d3d10umddi.h")
  message(FATAL_ERROR "The lab requires WDK user-mode headers")
endif()
add_library(nr_fill_umd SHARED driver.cpp)
target_include_directories(nr_fill_umd PRIVATE "${NR_WDK_INCLUDE_ROOT}/um" "${NR_WDK_INCLUDE_ROOT}/shared" "${NR_WDK_INCLUDE_ROOT}/km")
target_compile_features(nr_fill_umd PRIVATE cxx_std_17)
target_compile_options(nr_fill_umd PRIVATE /W4 /WX)
set_property(TARGET nr_fill_umd PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(nr_registered_clear_probe probe.cpp)
target_compile_features(nr_registered_clear_probe PRIVATE cxx_std_17)
target_compile_options(nr_registered_clear_probe PRIVATE /W4 /WX)
target_link_libraries(nr_registered_clear_probe PRIVATE d3d11 dxgi gdi32)
set_property(TARGET nr_registered_clear_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(nr_registered_lab_test tests.cpp)
target_include_directories(nr_registered_lab_test PRIVATE "${NR_WDK_INCLUDE_ROOT}/um" "${NR_WDK_INCLUDE_ROOT}/shared" "${NR_WDK_INCLUDE_ROOT}/km")
target_compile_features(nr_registered_lab_test PRIVATE cxx_std_17)
target_compile_options(nr_registered_lab_test PRIVATE /W4 /WX)
set_property(TARGET nr_registered_lab_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_test(NAME registered_lab_contract COMMAND nr_registered_lab_test)
