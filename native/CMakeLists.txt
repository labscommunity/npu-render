cmake_minimum_required(VERSION 3.25)
project(npu_render_native VERSION 0.2.0 LANGUAGES C CXX)

find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
add_library(npu_render_native SHARED src/backend.cpp)
target_include_directories(npu_render_native PUBLIC include)
target_compile_features(npu_render_native PRIVATE cxx_std_17)
target_link_libraries(npu_render_native PRIVATE openvino::runtime)
target_compile_definitions(npu_render_native PRIVATE NR_BUILD)
set_target_properties(npu_render_native PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_executable(nr_sampling_bench tools/sampling_bench.cpp)
target_compile_features(nr_sampling_bench PRIVATE cxx_std_17)
target_link_libraries(nr_sampling_bench PRIVATE npu_render_native)
add_executable(nr_resize_bench tools/sampling_bench.cpp)
target_compile_features(nr_resize_bench PRIVATE cxx_std_17)
target_compile_definitions(nr_resize_bench PRIVATE NR_BENCH_RESIZE)
target_link_libraries(nr_resize_bench PRIVATE npu_render_native)

if(WIN32)
  add_executable(nr_d3d12_interop tools/d3d12_interop.cpp)
  target_compile_features(nr_d3d12_interop PRIVATE cxx_std_17)
  target_link_libraries(nr_d3d12_interop PRIVATE openvino::runtime d3d12 dxgi)
  add_executable(nr_native_live windows/live.cpp)
  target_include_directories(nr_native_live PRIVATE include)
  target_compile_features(nr_native_live PRIVATE cxx_std_17)
  target_link_libraries(nr_native_live PRIVATE openvino::runtime d3d12 dxgi d3dcompiler user32)
endif()

include(CTest)
option(NR_WITH_PIXMAN "Build the restricted Weston adapter tests" OFF)
if(BUILD_TESTING)
  add_executable(nr_selection_test tests/selection_test.cpp)
  target_include_directories(nr_selection_test PRIVATE include)
  target_compile_features(nr_selection_test PRIVATE cxx_std_17)
  add_test(NAME native_selection COMMAND nr_selection_test)
  add_executable(nr_native_test tests/backend_test.cpp)
  target_compile_features(nr_native_test PRIVATE cxx_std_17)
  target_link_libraries(nr_native_test PRIVATE npu_render_native)
  add_test(NAME native_cpu COMMAND nr_native_test CPU)
  add_executable(nr_sampling_test tests/sampling_test.cpp)
  target_compile_features(nr_sampling_test PRIVATE cxx_std_17)
  target_link_libraries(nr_sampling_test PRIVATE npu_render_native)
  add_test(NAME native_sampling_cpu COMMAND nr_sampling_test CPU)
  add_executable(nr_resize_test tests/resize_test.cpp)
  target_compile_features(nr_resize_test PRIVATE cxx_std_17)
  target_link_libraries(nr_resize_test PRIVATE npu_render_native)
  add_test(NAME native_resize_cpu COMMAND nr_resize_test CPU)
endif()

if(NR_WITH_PIXMAN)
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  pkg_check_modules(PIXMAN REQUIRED IMPORTED_TARGET pixman-1)
  add_executable(nr_weston_test tests/weston_test.c weston/npu-render-weston.c)
  target_include_directories(nr_weston_test PRIVATE include weston)
  target_link_libraries(nr_weston_test PRIVATE PkgConfig::PIXMAN Threads::Threads ${CMAKE_DL_LIBS})
  add_test(NAME weston_adapter_cpu COMMAND nr_weston_test $<TARGET_FILE:npu_render_native>)
endif()

install(TARGETS npu_render_native LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/npu_render.h DESTINATION include)
