cmake_minimum_required(VERSION 3.25)
project(npu_native_admission_contract LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "x64 Windows MSVC required")
endif()
add_executable(nr_native_admission_contract test.cpp)
target_compile_features(nr_native_admission_contract PRIVATE cxx_std_17)
target_compile_options(nr_native_admission_contract PRIVATE /W4 /WX)
target_link_libraries(nr_native_admission_contract PRIVATE advapi32)
set_property(TARGET nr_native_admission_contract PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME native_admission_contract COMMAND nr_native_admission_contract)
set_tests_properties(native_admission_contract PROPERTIES TIMEOUT 30)
