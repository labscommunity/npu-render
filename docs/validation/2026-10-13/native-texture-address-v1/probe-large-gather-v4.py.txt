from pathlib import Path
import openvino as ov,numpy as np,json,os,gc,time,traceback
from openvino import opset13 as op
r=Path(os.environ['LOCALAPPDATA'])/'npu-render/vm/large-gather-v4';r.mkdir(exist_ok=False);results=[]
for n in (1048576,2097152,2097160,5311488):
 record={'elements':n,'passed':False};started=time.perf_counter()

  source=op.parameter([n,1],np.float32,name='source');indices=op.parameter([8],np.int32,name='indices');gather=op.gather(source,indices,op.constant(0,np.int32));model=ov.Model([gather],[source,indices])

  ov.serialize(model,str(r/f'model-{n}.xml'),str(r/f'model-{n}.bin'))

  values=((np.arange(n,dtype=np.uint32)*17+13)%251).astype(np.float32).reshape(n,1)/np.float32(255)

  record.update(passed=bool(np.array_equal(actual,expected)),maximum_error=float(np.max(np.abs(actual-expected))),indices=positions.tolist(),actual=actual.tolist(),expected=expected.tolist(),execution_devices=list(devices))
  del request,compiled,model,core;gc.collect()
