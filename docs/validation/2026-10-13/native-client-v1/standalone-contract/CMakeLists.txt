cmake_minimum_required(VERSION 3.25)
project(npu_local_client_contract LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "x64 Windows MSVC required")
endif()
if(NOT EXISTS "${NR_CLIENT_SOURCE}/local_indexed_client.hpp")
 message(FATAL_ERROR "Qualified client source required")
endif()
add_executable(nr_local_client_contract test.cpp)
target_include_directories(nr_local_client_contract PRIVATE "${NR_CLIENT_SOURCE}")
target_compile_features(nr_local_client_contract PRIVATE cxx_std_17)
target_compile_options(nr_local_client_contract PRIVATE /W4 /WX)
set_property(TARGET nr_local_client_contract PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME local_client_contract COMMAND nr_local_client_contract)
set_tests_properties(local_client_contract PROPERTIES TIMEOUT 30)
