cmake_minimum_required(VERSION 3.25)
project(pipeline_cache_diagnostic LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Qualified registered source")
set(NR_RENDER_LIBRARY "" CACHE FILEPATH "Qualified renderer library")
set(NR_CACHE_LIBRARY "" CACHE FILEPATH "Qualified cache library")
foreach(capacity 4 16)
 set(target pipeline_cache_${capacity})
 add_executable(${target} indexed_exchange_worker.cpp)
 target_compile_definitions(${target} PRIVATE NR_TRIANGLE_CACHE_CAPACITY=${capacity})
 target_include_directories(${target} PRIVATE "${NR_RENDER_SOURCE}")
 target_link_libraries(${target} PRIVATE "${NR_RENDER_LIBRARY}" "${NR_CACHE_LIBRARY}" openvino::runtime bcrypt)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE /W4 /WX)
 set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
