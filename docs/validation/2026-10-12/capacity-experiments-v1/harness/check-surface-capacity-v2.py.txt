import hashlib,json,struct,subprocess,sys
from pathlib import Path

def digest(b):return hashlib.sha256(b).hexdigest()
if mode=='prepare':
 build=Path(sys.argv[3]);group=sys.argv[4];assert group in ('targets','textures');out.mkdir(exist_ok=False);base=out/'base.bin'
 with (out/'fixture.log').open('wb') as f:subprocess.run([str(build/'nr_registered_indexed_exchange_test.exe'),'--emit-point-texture',str(base)],stdout=f,stderr=subprocess.STDOUT,check=True,timeout=30)

  clip=(3,5,w-7,h-9);color=bytes((17+(case%3)*31,71+(case%5)*17,129-(case%7)*11,case%256));old=bytes((3,19,91,137))
  body=struct.pack('<6f',0,0,w,h,0,1)+struct.pack('<4i',*clip)+original[104:inputs]+original[inputs:background]*quads+old*(w*h)+struct.pack('<4I',sw,sh,87,0)+color*(sw*sh)

  name=f'request-42-{hd[1]:016x}{hd[2]:016x}-{hd[11]}.bin';(out/name).write_bytes(data)
  records.append({'file':name,'sequence':hd[11],'case':case,'group':group,'width':w,'height':h,'vertices':6*quads,'texture_width':sw,'clip':clip,'background':list(old),'expected_color':list(color[:3]+b'\xff'),'covered_samples':(clip[2]-clip[0])*(clip[3]-clip[1])*quads,'request_sha256':digest(data)})

else:raise ValueError(mode)
