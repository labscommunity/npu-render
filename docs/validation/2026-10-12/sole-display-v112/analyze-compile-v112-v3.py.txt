import json,collections
root=Path('results/render-driver')
for trial in ['v112']:
 unmatched=[];rows=[json.loads(x) for x in (root/f'indexed-dwm-stream-{trial}/worker.stdout.jsonl').read_text().splitlines() if x.startswith('{')];success=[]
 for row in rows:
  if row.get('passed'):success.append({**row,'cache':None})
  elif row.get('kind')=='model_compile_cache':
   last=success[-1]
   if (last['client_pid'],last['sequence'])==(row['client_pid'],row['sequence']) and last['cache'] is None:last['cache']=row
   else:unmatched.append(row)
 live=[x for x in success if x['client_pid']!=42];out={}
 for name,pred in [('uniform',lambda x:x['operation']=='draw' and not x['textured']),('triangle',lambda x:x['triangle_batch']>0),('point_or_linear',lambda x:x['operation']=='draw' and x['textured'] and not x['triangle_batch']),('clear',lambda x:x['operation']=='color_clear')]:
  a=[x for x in live if pred(x)];comp=[x for x in a if x['cache'] is not None];plain=[x for x in a if x['cache'] is None];m=[x['cache'] for x in comp];out[name]={'requests':len(a),'requests_compiling_models':len(comp),'compile_calls':sum(x['compile_calls'] for x in m),'persistent_cache_hits':sum(x['cache_hits'] for x in m),'draw_ms':round(sum(x['draw_ms'] for x in a),3),'compile_and_draw_ms':sum(x['compile_and_draw_ms'] for x in a),'outside_draw_with_compile_ms':round(sum(x['compile_and_draw_ms']-x['draw_ms'] for x in comp),3),'outside_draw_without_compile_ms':round(sum(x['compile_and_draw_ms']-x['draw_ms'] for x in plain),3)}
 assert sum(x['requests'] for x in out.values())==len(live)
 report={'method':'Attach compile metadata in stream order to its preceding successful result, allowing slow-packet records between them and checking PID/sequence. Retain metadata without a successful result separately; verify it against a negative relay receipt. Keys alone are not unique.', 'metadata_without_success':unmatched,'duplicate_pid_sequence_keys':sum(v>1 for v in collections.Counter((x['client_pid'],x['sequence']) for x in live).values()),'categories':out}
 failed=[x for x in [json.loads(a) for a in (root/f'indexed-dwm-stream-{trial}/relay.stdout.jsonl').read_text().splitlines() if a.startswith('{')] if x.get('worker_success') is False]
 assert collections.Counter((x['client_pid'],x['sequence']) for x in unmatched)==collections.Counter((x['client_pid'],x['sequence']) for x in failed)
 (root/f'{trial}-compile-event-audit-v3.json').write_text(json.dumps(report,indent=2)+'\n');print(trial,out['triangle'])
