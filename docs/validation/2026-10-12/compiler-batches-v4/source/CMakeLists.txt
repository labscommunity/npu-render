cmake_minimum_required(VERSION 3.25)
project(compiler_batch_diagnostic LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Registered source")
set(NR_RENDER_LIBRARY "" CACHE FILEPATH "Qualified renderer library")
set(NR_CACHE_LIBRARY "" CACHE FILEPATH "Provider cache library")
add_executable(nr_compositor_batch_probe compositor_batch_probe.cpp)
target_include_directories(nr_compositor_batch_probe PRIVATE "${NR_RENDER_SOURCE}")
target_link_libraries(nr_compositor_batch_probe PRIVATE "${NR_RENDER_LIBRARY}" "${NR_CACHE_LIBRARY}" openvino::runtime bcrypt)
target_compile_features(nr_compositor_batch_probe PRIVATE cxx_std_17)
target_compile_options(nr_compositor_batch_probe PRIVATE /W4 /WX)
set_property(TARGET nr_compositor_batch_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
