import sys,json,time,hashlib
from datetime import datetime,UTC
from pathlib import Path
sys.path.insert(0,str(Path.cwd()))
from tools.view_lab_desktop import LabInput
from npu_render.indirect_frame import confirmed_sole_display,parse_frame
root=Path('results/render-driver');out=root/'input-v111';out.mkdir(exist_ok=False)
source=root/'collector-v111-live/latest.bin';gate=source.parent/'input-ready.json';control=LabInput(69629,49400)
report={'started':datetime.now(UTC).isoformat(),'actions':[],'snapshots':[],'visual_response_verified':False,'sole_gate_observed':False}
def persist():
 (out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
def snapshot(name):
 if source.exists():
  data=source.read_bytes();frame=parse_frame(data);(out/(name+'.bin')).write_bytes(data)
  report['snapshots'].append({'name':name,'captured_at':datetime.now(UTC).isoformat(),'sole_indirect_at_snapshot':confirmed_sole_display(gate),'identity':frame.identity,'pixel_sha256':hashlib.sha256(frame.pixels).hexdigest()})
try:
 deadline=time.monotonic()+650;woke=False;sole_at=None
 while time.monotonic()<deadline:
  if confirmed_sole_display(gate):
   report['sole_gate_observed']=True
   if sole_at is None:sole_at=time.monotonic()
   if not woke:
    control.key('ctrl',True);time.sleep(.2);control.key('ctrl',False);woke=True;report['actions'].append({'action':'wake-control','at':datetime.now(UTC).isoformat()})
   if not source.exists():
    time.sleep(.5);continue
   candidate=parse_frame(source.read_bytes())
   receipts=[]
   for line in (source.parent/'frames.jsonl').read_text().splitlines():
    try:receipts.append(json.loads(line))
    except json.JSONDecodeError:pass
   if not any(tuple(r['identity'])==candidate.identity and r['sole_indirect_at_receive'] for r in receipts):
    time.sleep(.5);continue
   # Wait for nonuniform visible content; a black wallpaper must not block input testing.
   if len({bytes(candidate.pixels[n:n+3]) for n in range(0,len(candidate.pixels),64)}) < 16:
    time.sleep(.5);continue
   # Input eligibility comes from verified topology and a received nonblack frame.
   # Taskbar color/transparency is not a readiness signal. Require a separately
   # inspected complete desktop frame before the first interaction. Stale
   # topology confirmation pauses input until a fresh confirmation arrives.
   if not (out/'desktop-ready.json').exists():
    time.sleep(.5);continue
   snapshot('before-start')
   control.key('meta_l',True);time.sleep(.2);control.key('meta_l',False);report['actions'].append({'action':'open-start','at':datetime.now(UTC).isoformat()})
   persist()
   while time.monotonic()<deadline and not ((out/'start-visible.json').exists() and confirmed_sole_display(gate)):
    time.sleep(.5)
   snapshot('after-start');persist()
   if (out/'start-visible.json').exists() and confirmed_sole_display(gate) and time.monotonic()<deadline:
    control.key('esc',True);time.sleep(.2);control.key('esc',False);report['actions'].append({'action':'escape','at':datetime.now(UTC).isoformat()})
    persist()
    while time.monotonic()<deadline and not ((out/'escape-visible.json').exists() and confirmed_sole_display(gate)):
     time.sleep(.5)
    snapshot('after-escape');persist()
    if (out/'escape-visible.json').exists() and confirmed_sole_display(gate) and time.monotonic()+30<deadline:
     control.key('meta_l',True);control.key('r',True);time.sleep(.2);control.key('r',False);control.key('meta_l',False)
     report['actions'].append({'action':'open-run','at':datetime.now(UTC).isoformat()});persist()
     for marker in ('run-visible','notepad-visible','text-visible','maximized-visible','closed-visible'):
      while time.monotonic()<deadline and not ((out/(marker+'.json')).exists() and confirmed_sole_display(gate)):time.sleep(.5)
      if not (out/(marker+'.json')).exists() or not confirmed_sole_display(gate):break
      snapshot(marker);persist()
      if marker=='run-visible':
       for code in 'notepad':control.key(code,True);time.sleep(.1);control.key(code,False);time.sleep(.15)
       control.key('ret',True);time.sleep(.2);control.key('ret',False)
       report['actions'].append({'action':'launch-notepad','at':datetime.now(UTC).isoformat()})
      elif marker=='notepad-visible':
       for character in 'npu desktop offload v111':
        code='spc' if character==' ' else character;control.key(code,True);time.sleep(.1);control.key(code,False);time.sleep(.15)
       report['actions'].append({'action':'type-test-text','text':'npu desktop offload v111','at':datetime.now(UTC).isoformat()})
      elif marker=='text-visible':
       control.key('meta_l',True);control.key('up',True);time.sleep(.2);control.key('up',False);control.key('meta_l',False)
       report['actions'].append({'action':'maximize-notepad','at':datetime.now(UTC).isoformat()})
      elif marker=='maximized-visible':
       control.key('alt',True);control.key('f4',True);time.sleep(.2);control.key('f4',False);control.key('alt',False)
       report['actions'].append({'action':'close-notepad','at':datetime.now(UTC).isoformat()})
      persist()

   break
  time.sleep(.5)
finally:
 control.release();report['finished']=datetime.now(UTC).isoformat();(out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
 print(json.dumps(report))
