cmake_minimum_required(VERSION 3.20)
project(nr_crop_transition LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(nr_crop_transition contract.cpp)
target_compile_features(nr_crop_transition PRIVATE cxx_std_17)
target_compile_options(nr_crop_transition PRIVATE /W4 /WX)
target_include_directories(nr_crop_transition PRIVATE "${NR_SOURCE}/native/windows/registered_lab")
target_link_libraries(nr_crop_transition PRIVATE bcrypt d3dcompiler "${NR_BUILD}/Release/nr_registered_vertex.lib" "${NR_BUILD}/Release/nr_registered_provider_cache.lib" openvino::runtime)
set_property(TARGET nr_crop_transition PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
