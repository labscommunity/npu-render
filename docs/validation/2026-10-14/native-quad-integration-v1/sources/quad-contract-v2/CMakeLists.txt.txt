cmake_minimum_required(VERSION 3.25)
project(quad_contract LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Requires x64 MSVC")
endif()
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
add_executable(nr_quad_pipeline pipeline.cpp)
target_include_directories(nr_quad_pipeline PRIVATE "${NR_BASE_SOURCE}/native/windows/registered_lab")
target_link_libraries(nr_quad_pipeline PRIVATE "${NR_BASE_BUILD}/Release/nr_registered_vertex.lib" "${NR_BASE_BUILD}/Release/nr_registered_provider_cache.lib" openvino::runtime d3dcompiler bcrypt)
target_compile_features(nr_quad_pipeline PRIVATE cxx_std_17)
target_compile_options(nr_quad_pipeline PRIVATE /W4 /WX)
set_property(TARGET nr_quad_pipeline PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
