cmake_minimum_required(VERSION 3.25)
project(npu_output_observer LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires x64 Windows/MSVC")
endif()
add_executable(nr_output_observer observer.cpp)
target_compile_features(nr_output_observer PRIVATE cxx_std_17)
target_compile_definitions(nr_output_observer PRIVATE UNICODE _UNICODE)
target_compile_options(nr_output_observer PRIVATE /W4 /WX)
target_link_libraries(nr_output_observer PRIVATE d3d11 dxgi user32 gdi32 wtsapi32 advapi32)
set_property(TARGET nr_output_observer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
