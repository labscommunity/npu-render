cmake_minimum_required(VERSION 3.20)
project(nr_point_crop LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(nr_point_crop contract.cpp)
target_compile_features(nr_point_crop PRIVATE cxx_std_17)
target_compile_options(nr_point_crop PRIVATE /W4 /WX)
target_include_directories(nr_point_crop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${NR_SOURCE}/native/windows/registered_lab")
target_link_libraries(nr_point_crop PRIVATE bcrypt "${NR_BUILD}/Release/nr_registered_provider_cache.lib" openvino::runtime)
set_property(TARGET nr_point_crop PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
