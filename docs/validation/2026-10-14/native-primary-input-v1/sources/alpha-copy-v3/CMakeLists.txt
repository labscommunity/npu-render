cmake_minimum_required(VERSION 3.20)
project(nr_alpha_copy LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(nr_alpha_copy contract.cpp)
target_compile_features(nr_alpha_copy PRIVATE cxx_std_17)
target_compile_options(nr_alpha_copy PRIVATE /W4 /WX)
target_include_directories(nr_alpha_copy PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${NR_SOURCE}/native/windows/registered_lab")
target_link_libraries(nr_alpha_copy PRIVATE bcrypt "${NR_BUILD}/Release/nr_registered_provider_cache.lib" openvino::runtime)
set_property(TARGET nr_alpha_copy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
