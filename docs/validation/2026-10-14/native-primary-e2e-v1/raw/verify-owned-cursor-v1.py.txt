"""Compare a controlled-window NPU frame patch with the current Windows cursor.

DrawIconEx supplies a CPU reference only; it does not alter or supply NPU pixels.
"""
from pathlib import Path
from datetime import datetime,UTC
from ctypes import wintypes as W
import ctypes as C,json,struct,sys,hashlib
from native_frame_io_v1 import open_shared_frame,k
u=C.WinDLL('user32',use_last_error=True);g=C.WinDLL('gdi32',use_last_error=True)
class Cursor(C.Structure):_fields_=[('size',W.DWORD),('flags',W.DWORD),('cursor',W.HANDLE),('position',W.POINT)]
class Icon(C.Structure):_fields_=[('is_icon',W.BOOL),('hot_x',W.DWORD),('hot_y',W.DWORD),('mask',W.HANDLE),('color',W.HANDLE)]
class Bitmap(C.Structure):_fields_=[('type',W.LONG),('width',W.LONG),('height',W.LONG),('stride',W.LONG),('planes',W.WORD),('bits_per_pixel',W.WORD),('bits',W.LPVOID)]
class Header(C.Structure):_fields_=[('size',W.DWORD),('width',W.LONG),('height',W.LONG),('planes',W.WORD),('bits',W.WORD),('compression',W.DWORD),('bytes',W.DWORD),('xppm',W.LONG),('yppm',W.LONG),('used',W.DWORD),('important',W.DWORD)]
class Info(C.Structure):_fields_=[('header',Header),('colors',W.DWORD*1)]
for dll,name,args,result in [(u,'GetCursorInfo',[C.POINTER(Cursor)],W.BOOL),(u,'GetIconInfo',[W.HANDLE,C.POINTER(Icon)],W.BOOL),(u,'SetThreadDpiAwarenessContext',[W.HANDLE],W.HANDLE),(u,'WindowFromPoint',[W.POINT],W.HWND),(u,'GetWindowThreadProcessId',[W.HWND,C.POINTER(W.DWORD)],W.DWORD),(u,'DrawIconEx',[W.HDC,C.c_int,C.c_int,W.HANDLE,C.c_int,C.c_int,W.UINT,W.HBRUSH,W.UINT],W.BOOL),(g,'GetObjectW',[W.HANDLE,C.c_int,W.LPVOID],C.c_int),(g,'CreateCompatibleDC',[W.HDC],W.HDC),(g,'CreateDIBSection',[W.HDC,C.POINTER(Info),W.UINT,C.POINTER(W.LPVOID),W.HANDLE,W.DWORD],W.HANDLE),(g,'SelectObject',[W.HDC,W.HANDLE],W.HANDLE),(g,'DeleteObject',[W.HANDLE],W.BOOL),(g,'DeleteDC',[W.HDC],W.BOOL),(g,'GdiFlush',[],W.BOOL),(k,'GetFileSizeEx',[W.HANDLE,C.POINTER(C.c_longlong)],W.BOOL),(k,'ReadFile',[W.HANDLE,W.LPVOID,W.DWORD,C.POINTER(W.DWORD),W.LPVOID],W.BOOL)]:
 f=getattr(dll,name);f.argtypes=args;f.restype=result
frame=Path(sys.argv[1]);producer=int(sys.argv[2]);input_pid=int(sys.argv[3]);out=Path(sys.argv[4]);assert not out.exists()
report=dict(started=datetime.now(UTC).isoformat(),completed=False,passed=False,screenshots_saved=False,cursor_movement_tested=False,physical_cursor_observed=False,reference_backend='Windows DrawIconEx into CPU DIB; comparison only')
icon=Icon();dc=None;dib=None;old=None;handle=None
try:
 assert u.SetThreadDpiAwarenessContext(W.HANDLE(-4))
 before=Cursor();before.size=C.sizeof(before);assert u.GetCursorInfo(C.byref(before)) and before.flags&1 and (before.position.x,before.position.y)==(300,180)
 owner=u.WindowFromPoint(before.position);pid=W.DWORD();u.GetWindowThreadProcessId(owner,C.byref(pid));assert pid.value==input_pid
 assert u.GetIconInfo(before.cursor,C.byref(icon));bitmap=Bitmap();assert g.GetObjectW(icon.color or icon.mask,C.sizeof(bitmap),C.byref(bitmap))==C.sizeof(bitmap)
 w=bitmap.width;h=bitmap.height if icon.color else bitmap.height//2;x=before.position.x-icon.hot_x;y=before.position.y-icon.hot_y
 assert 0<w<=256 and 0<h<=256 and 64<=x and 64<=y and x+w<=576 and y+h<=320
 handle=open_shared_frame(frame);assert handle;size=C.c_longlong();assert k.GetFileSizeEx(handle,C.byref(size)) and size.value==64+1024*768*4
 data=C.create_string_buffer(size.value);n=W.DWORD();assert k.ReadFile(handle,data,size.value,C.byref(n),None) and n.value==size.value
 fields=struct.unpack('<QIIQIIIIQQII',data.raw[:64]);assert fields[0]==0x31454d415246524e and fields[1:3]==(1,64) and fields[4:9]==(1024,768,87,4096,3145728) and fields[10]==producer and fields[11]==0
 pixels=data.raw[64:];bg=pixels[(70*1024+70)*4:(70*1024+70)*4+4];assert bg in (bytes([180,60,30,255]),bytes([60,180,20,255]),bytes([80,30,180,255]))
 info=Info();info.header=Header(C.sizeof(Header),w,-h,1,32,0,w*h*4,0,0,0,0);bits=W.LPVOID();dc=g.CreateCompatibleDC(None);assert dc;dib=g.CreateDIBSection(dc,C.byref(info),0,C.byref(bits),None,0);assert dib and bits.value
 old=g.SelectObject(dc,dib);assert old;C.memmove(bits,bg*(w*h),w*h*4);assert u.DrawIconEx(dc,0,0,before.cursor,w,h,0,None,3) and g.GdiFlush()
 expected=C.string_at(bits,w*h*4);actual=b''.join(pixels[((y+row)*1024+x)*4:((y+row)*1024+x+w)*4] for row in range(h))
 after=Cursor();after.size=C.sizeof(after);assert u.GetCursorInfo(C.byref(after));after_owner=u.WindowFromPoint(after.position);after_pid=W.DWORD();u.GetWindowThreadProcessId(after_owner,C.byref(after_pid))
 assert after.cursor==before.cursor and after.flags==before.flags and (after.position.x,after.position.y)==(300,180) and after_owner==owner and after_pid.value==input_pid
 errors=[abs(a-b) for i,(a,b) in enumerate(zip(actual,expected)) if i%4!=3];changed=sum(expected[i:i+3]!=bg[:3] for i in range(0,len(expected),4));assert changed>0
 report.update(sequence=fields[3],qpc=fields[9],producer=producer,cursor_width=w,cursor_height=h,cursor_hotspot=[icon.hot_x,icon.hot_y],cursor_position=[300,180],controlled_background=list(bg),reference_changed_pixels=changed,maximum_rgb_error=max(errors),different_rgb_components=sum(e!=0 for e in errors),npu_patch_sha256=hashlib.sha256(actual).hexdigest(),cpu_reference_sha256=hashlib.sha256(expected).hexdigest(),passed=max(errors)==0,ownership_and_cursor_unchanged_during_comparison=True)
except Exception as e:report['error']=repr(e)
finally:
 if handle:k.CloseHandle(handle)
 if old and dc:g.SelectObject(dc,old)
 if dib:g.DeleteObject(dib)
 if dc:g.DeleteDC(dc)
 for bitmap in (icon.color,icon.mask):
  if bitmap:g.DeleteObject(bitmap)
 report.update(completed=True,finished=datetime.now(UTC).isoformat());out.write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report));raise SystemExit(0 if report['passed'] else 1)
