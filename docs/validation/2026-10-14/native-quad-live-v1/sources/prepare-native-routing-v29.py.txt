from pathlib import Path
import hashlib,json,shutil,tarfile
r=Path('results/render-driver');d=r/'native-routing-v29';shutil.copytree(r/'native-routing-v28',d)
p=d/'native/windows/registered_lab/opaque_affine_crop.hpp';s=p.read_text()
s=s.replace('uint32_t sw,sh,w,h;bool', 'uint32_t sw,sh,w,h,band_pixels=65535;bool')
anchor='        if(!sw || sw>8192'
assert s.count(anchor)==1
s=s.replace(anchor,'''        char *setting=nullptr;size_t length=0;
        if(_dupenv_s(&setting,&length,"NR_RENDER_CROP_BAND"))throw std::runtime_error("cannot read crop band experiment");
        if(setting){const std::string value(setting);std::free(setting);
            if(value=="65535")band_pixels=65535;else if(value=="262143")band_pixels=262143;else if(value=="1048575")band_pixels=1048575;else throw std::invalid_argument("invalid crop band experiment");}
'''+anchor)
assert s.count('65535u/(std::max)(sw,w)')==1
s=s.replace('65535u/(std::max)(sw,w)','band_pixels/(std::max)(sw,w)');p.write_text(s)
(r/'native-routing-v29-source.json').write_text(json.dumps({p.relative_to(d).as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in d.rglob('*') if p.is_file()},indent=2)+'\n')
with tarfile.open(r/'native-routing-v29.tar.gz','w:gz') as t:t.add(d,arcname=d.name)
(r/'build-native-routing-v29.ps1').write_text((r/'build-native-routing-v28.ps1').read_text().replace('native-routing-v28','native-routing-v29'))
s=(r/'qualify-native-quad-v2.py').read_text().replace('native-routing-v28','native-routing-v29')
s=s.replace("p.add_argument('--tickets',required=True)","p.add_argument('--band',choices=('65535','262143','1048575'),required=True);p.add_argument('--tickets',required=True)")
s=s.replace("live_performance_qualified=False,", "live_performance_qualified=False,crop_band=v.band,")
s=s.replace("env.update(NR_RENDER_OPAQUE_CROP='1',", "env.update(NR_RENDER_OPAQUE_CROP='1',NR_RENDER_CROP_BAND=v.band,")
(r/'qualify-native-crop-band-v1.py').write_text(s)
print('Prepared opt-in crop band candidate; default behavior unchanged, hardware unqualified.')
