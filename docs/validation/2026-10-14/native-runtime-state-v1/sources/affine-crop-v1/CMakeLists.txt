cmake_minimum_required(VERSION 3.20)
project(nr_affine_crop LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
add_executable(nr_affine_crop probe.cpp linear_texture_draw.cpp)
target_compile_features(nr_affine_crop PRIVATE cxx_std_17)
target_compile_options(nr_affine_crop PRIVATE /W4 /WX)
target_include_directories(nr_affine_crop PRIVATE "${NR_SOURCE}/native/windows/registered_lab" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(nr_affine_crop PRIVATE d3dcompiler bcrypt "${NR_BUILD}/Release/nr_registered_vertex.lib" "${NR_BUILD}/Release/nr_registered_provider_cache.lib" openvino::runtime)
set_property(TARGET nr_affine_crop PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
