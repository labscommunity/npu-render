"""Lower the bounded straight-line shader to auditable scalar SSA C.

This is an experimental source generator, not an application admission path.
Separate coordinate and color functions enable NPU-generated texture gathering.
"""
from pathlib import Path
import hashlib
import json
import re

import argparse
parser=argparse.ArgumentParser();parser.add_argument('--ir',type=Path,required=True);parser.add_argument('--output',type=Path,required=True);args=parser.parse_args()
out=args.output;out.mkdir(exist_ok=False)
ir=json.loads(args.ir.read_text())['pixel']
assert 0<len(ir['instructions'])<=256 and ir['temporaries']<=16
nodes = {}
registers = {}
coordinates = []
samples = []

def emit(expression):
    name = 's' + str(len(nodes))
    nodes[name] = expression
    return name

def source(o, lane, integer=False):
    assert not o['relative']
    c = o['lanes'][lane]
    if o['kind'] == 0 or o['kind'] == 2:
        value = registers[(o['kind'], o['index'], c)]
    elif o['kind'] == 1:
        assert o['index'] < 8
        value = f'input[{o["index"] * 4 + c}]'
    elif o['kind'] == 8:
        assert o['index'] == 0 and o['element'] < 64
        value = f'constant[{o["element"] * 4 + c}]'
    elif o['kind'] == 4:
        value = str(o['literal'][c]) + 'u'
    else:
        raise ValueError(o)
    if o['modifier'] in (2, 3):
        value = f'({value}&0x7fffffffu)'
    if o['modifier'] in (1, 3):
        value = f'(0u-{value})' if integer else f'({value}^0x80000000u)'
    return value

for instruction in ir['instructions']:
    op = instruction['opcode']
    d = instruction['destination']
    assert d['kind'] in (0, 2)
    src = instruction['sources']
    sampled = None
    if op == 69:
        sampled = len(samples)
        coordinates.extend([source(src[0], 0), source(src[0], 1)])
        assert src[1]['index'] == src[2]['index']
        samples.append(src[1]['index'])
    writes = {}
    for c in range(4):
        if not d['mask'] & (1 << c):
            continue
        if op == 69:
            expression = f'sampled[{sampled * 4 + src[1]["lanes"][c]}]'
        else:
            a = [source(s, c, op == 30) for s in src]
            f = [f'nr_float({v})' for v in a]
            if op == 54: expression = a[0]
            elif op == 55: expression = f'({a[0]}?{a[1]}:{a[2]})'
            elif op == 30: expression = f'({a[0]}+{a[1]})'
            elif op == 1: expression = f'({a[0]}&{a[1]})'
            elif op == 43: expression = f'nr_bits((float)(int){a[0]})'
            elif op in (49, 29, 24):
                sign = {49: '<', 29: '>=', 24: '=='}[op]
                values = f if op == 49 else [f'nr_flush({v})' for v in f]
                expression = f'({values[0]}{sign}{values[1]}?0xffffffffu:0u)'
            else:
                if op == 0: value = f'({f[0]}+{f[1]})'
                elif op == 56: value = f'({f[0]}*{f[1]})'
                elif op == 14: value = f'({f[0]}/{f[1]})'
                elif op == 50: value = f'(({f[0]}*{f[1]})+{f[2]})'
                elif op == 51: value = f'nr_min({f[0]},{f[1]})'
                elif op == 52: value = f'nr_max({f[0]},{f[1]})'
                elif op == 65: value = f'nr_floor({f[0]})'
                elif op == 26: value = f'nr_fraction({f[0]})'
                elif op == 16:
                    product = [f'(nr_float({source(src[0], k)})*nr_float({source(src[1], k)}))' for k in range(3)]
                    value = f'(({product[0]}+{product[1]})+{product[2]})'
                else: raise ValueError(op)
                expression = f'nr_bits({value})'
        if instruction['saturate']:
            expression = f'nr_bits(nr_saturate(nr_float({expression})))'
        writes[(d['kind'], d['index'], c)] = emit(expression)
    registers.update(writes)

def lower(name, outputs, target, allow_samples):
    selected = set()
    def visit(value):
        for n in re.findall(r'\bs\d+\b', value):
            if n not in selected:
                selected.add(n)
                visit(nodes[n])
    for v in outputs: visit(v)
    if not allow_samples:
        assert all('sampled[' not in nodes[n] for n in selected)
    lines = [f'INLINE void {name}(const unsigned *input,const unsigned *constant,const unsigned *sampled,unsigned *{target})' + '{']
    lines += [f' unsigned {n}={expr};' for n, expr in nodes.items() if n in selected]
    lines += [f' {target}[{c}]={v};' for c, v in enumerate(outputs)]
    lines += ['}']
    return '\n'.join(lines), len(selected)

helpers = r'''/* Generated structurally lowered shader experiment. No host rasterization. */
#pragma STDC FP_CONTRACT OFF
#ifndef INLINE
#define INLINE static __attribute__((always_inline)) inline
#endif
INLINE float nr_float(unsigned v){union{unsigned u;float f;}x;x.u=v;return x.f;}
INLINE unsigned nr_bits(float v){union{unsigned u;float f;}x;x.f=v;return x.u;}
INLINE float nr_flush(float v){return (nr_bits(v)&0x7f800000u)?v:0.0f;}
INLINE float nr_min(float a,float b){if(a!=a)return b;if(b!=b)return a;return a<b?a:b;}
INLINE float nr_max(float a,float b){if(a!=a)return b;if(b!=b)return a;return a<b?b:a;}
INLINE float nr_saturate(float a){a=nr_flush(a);return !(a>=0)?0:(a>1?1:a);}
INLINE float nr_floor(float a){
 unsigned bits=nr_bits(a),mag=bits&0x7fffffffu;
 if(mag<0x00800000u)return nr_float(bits&0x80000000u);
 if(mag>=0x4b000000u)return a;
 int i=(int)a;float b=(float)i;return b>a?b-1.0f:b;
}
INLINE float nr_fraction(float a){return (nr_bits(a)&0x7fffffffu)<0x00800000u?0.0f:a-nr_floor(a);}
'''
plan, pn = lower('nr_coordinates', coordinates, 'coordinates', False)
color, cn = lower('nr_color', [registers[(2, 0, c)] for c in range(4)], 'color', True)
source_text = helpers + plan + '\n' + color + '\n'
(out / 'shader.h').write_text(source_text)
report = dict(cpu_codegen_only=True, sample_resources=samples, independent_sample_coordinates=True,
              coordinate_scalar_operations=pn, color_scalar_operations=cn,
              ir_sha256=hashlib.sha256(args.ir.read_bytes()).hexdigest(),
              shader_sha256=hashlib.sha256(source_text.encode()).hexdigest())
(out / 'report.json').write_text(json.dumps(report, indent=2)+'\n')
print(json.dumps(report))
