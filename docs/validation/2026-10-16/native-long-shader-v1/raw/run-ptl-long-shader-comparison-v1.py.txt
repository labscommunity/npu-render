"""Compare the timed-out desktop request with the qualified long-shader component."""
from pathlib import Path
from datetime import UTC, datetime
import hashlib
import json
import os
import subprocess
import traceback

r=Path(os.environ['LOCALAPPDATA'])/'npu-render';vm=r/'vm'
out=vm/'ptl-long-shader-comparison-v1';out.mkdir(exist_ok=False)
def obj(p):
    b=p.read_bytes();return json.loads(b.decode('utf-16') if b.startswith(b'\xff\xfe') else b.decode('utf-8-sig'))
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
report=dict(started=datetime.now(UTC).isoformat(),completed=False,passed=False,desktop_changed=False,tahoma_tested=False,original_successful_reference_available=False)
process=None
def save():(out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
save()
try:
    assert obj(vm/'ptl-six-span-long-shader-v1/report.json')['passed']
    assert obj(vm/'native-desktop-admission-v53/report.json')['cleanup_passed']
    model=vm/'ptl-six-span-long-shader-v1/patched.blob';assert sha(model)=='f12fd4a8622fbe1ccd122163088067dcd8aba5066ce56151f6f90a3556920592'
    geometry=vm/'ptl-six-span-geometry-v2/patched.blob';assert sha(geometry)=='ea4432abd7b548396b86665c1b318353e0d2f3ec847bcd3e6f56c10bd1336dfa'
    exe=r/'builds/ptl-long-shader-integration-v1/Release/nr_ptl_four_tap_compare.exe';assert sha(exe)=='764383894c88efbcada485ae5d31209a36123a062d6863c1d56887c6afef3329'
    packet=vm/'native-desktop-admission-v53/bootstrap/failed-request.bin';assert sha(packet)=='46caf6546bcde449d3829a05aaedfcefc719686f25264cec6786065df7a628c3'
    env=os.environ.copy();env['PATH']=str(r/'toolchains/openvino-2026.4.1/Lib/site-packages/openvino/libs')+os.pathsep+env['PATH']
    env.update(NR_RENDER_NPU_CACHE=str(vm/'model-cache-2026-4-v209'),NR_RENDER_SHARED_QUEUE='1',NR_RENDER_MEMORY_CACHE='1',NR_RENDER_MEMORY_CACHE_CAPACITY='512',NR_RENDER_SHARED_CORE='1',NR_RENDER_OPAQUE_CROP='1',NR_RENDER_TEXTURE_SCOPE='1',NR_RENDER_QUAD_BATCH='1',NR_RENDER_COMPACT_COVERAGE='0',NR_RENDER_INFER_TRACE='0',NR_RENDER_COMPILE_TRACE='1',NR_RENDER_INFER_PROFILE='0',NR_RENDER_PTL_LONG_BLOB=str(model),NR_RENDER_PTL_GEOMETRY_BLOB=str(geometry))
    env.pop('NR_RENDER_AFFINE_BATCH',None)
    report.update(model_sha256=sha(model),executable_sha256=sha(exe),packet_sha256=sha(packet),offline_deadline_seconds=300,desktop_deadline_changed=False);save()
    with (out/'stdout.jsonl').open('wb') as stdout,(out/'stderr.log').open('wb') as stderr:
        process=subprocess.Popen([str(exe),str(packet),str(model),str(out)],env=env,stdout=stdout,stderr=stderr,creationflags=subprocess.CREATE_NO_WINDOW)
        try:process.wait(timeout=300)
        except subprocess.TimeoutExpired:process.kill();process.wait(timeout=5);report['child_timeout']=True
    events=[json.loads(x) for x in (out/'stdout.jsonl').read_text().splitlines() if x.startswith('{') and x.endswith('}')]
    pairs=[x for x in events if x.get('kind')=='pair']
    report.update(exit_code=process.returncode,pairs=pairs,pixel_artifacts={p.name:dict(bytes=p.stat().st_size,sha256=sha(p)) for p in out.glob('*.rgba')})
    assert process.returncode==0 and len(pairs)==4 and all(x['exact'] for x in pairs)
    assert any(x.get('kind')=='complete' and x.get('passed') for x in events)
    report['passed']=True
except Exception as error:report.update(error=repr(error),traceback=traceback.format_exc())
finally:
    if process and process.poll() is None:process.kill();process.wait(timeout=5)
    report.update(completed=True,finished=datetime.now(UTC).isoformat());save()
print(json.dumps(report));raise SystemExit(0 if report['passed'] else 1)
