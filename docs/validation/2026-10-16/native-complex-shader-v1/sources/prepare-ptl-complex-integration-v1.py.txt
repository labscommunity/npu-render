"""Prepare full-image candidate only after a model digest is available."""
from pathlib import Path
import argparse,hashlib,json,re,shutil,struct,tarfile
p=argparse.ArgumentParser();p.add_argument('--model-sha256',required=True);args=p.parse_args();assert re.fullmatch('[0-9a-f]{64}',args.model_sha256)
r=Path('results/render-driver');d=r/'ptl-complex-integration-v1';d.mkdir(exist_ok=False)
for source in (r/'ptl-long-shader-integration-v4').iterdir():
 if source.is_file():
  s=source.read_text().replace('PtlLong','PtlComplex').replace('ptl_long','ptl_complex').replace('PTL long','PTL complex').replace('long shader comparison','complex shader comparison').replace('NR_RENDER_PTL_LONG_BLOB','NR_RENDER_PTL_COMPLEX_BLOB').replace('nr_ptl_four_tap_compare','nr_ptl_complex_compare')
  (d/source.name.replace('ptl_long','ptl_complex')).write_text(s)
header=Path('docs/validation/2026-10-16/native-shader-inventory-v1/raw/captured.hpp').read_text();found=[]
for name,body in re.findall(r'inline constexpr uint32_t (pixel\d+)\[\]\{([^}]+)\};',header):
 words=[int(n.rstrip('u'),16) for n in body.split(',')];digest=hashlib.sha256(struct.pack('<'+'I'*len(words),*words)).hexdigest()
 if digest=='c71638aa9df382391c81ce814b60b92cff815bf2b31547a5569cfe785beaf341':found.append(words)
assert len(found)==1
(d/'ptl_complex_shaders.hpp').write_text('#pragma once\n#include <array>\nnamespace nr::shader::ptl_complex_fixture {\ninline constexpr std::array<unsigned,'+str(len(found[0]))+'> pixel{'+','.join(hex(x)+'u' for x in found[0])+'};\n}\n')
p=d/'ptl_complex_shader.hpp';s=p.read_text();s=s.replace('unsigned count=0,address_u=0,address_v=0;','unsigned count=0,address_u[2]{},address_v[2]{},filters[2]{};').replace('if(n<1)','if(n<2)').replace('s.filter!=TextureFilter::linear','(s.filter!=TextureFilter::linear&&s.filter!=TextureFilter::point)')
a=s.index('  constexpr unsigned char expected[]{');b=s.index(';',a)
s=s[:a]+'  constexpr unsigned char expected[]{'+','.join('0x'+args.model_sha256[n:n+2] for n in range(0,64,2))+'}'+s[b:]
s=s.replace('address_u=unsigned(samplers[0].u);address_v=unsigned(samplers[0].v);','for(unsigned n=0;n<2;++n){address_u[n]=unsigned(samplers[n].u);address_v[n]=unsigned(samplers[n].v);filters[n]=samplers[n].filter==TextureFilter::point?1u:0u;}')
s=s.replace('constants[0].size()>=10','constants[0].size()>=6').replace('n<1;','n<2;').replace('first+=3072','first+=6144').replace('plane*512','plane*1024').replace('(512u,total-begin)','(1024u,total-begin)').replace('0x4e524c31','0x4e524331').replace('0x4e524c4f','0x4e52434f').replace('put(p+4,2)','put(p+4,1)')
s=s.replace('put(p+32,address_u);put(p+36,address_v);std::memcpy(p+64,constants[0].data(),160);','for(unsigned n=0;n<2;++n){put(p+32+n*8,address_u[n]);put(p+36+n*8,address_v[n]);put(p+48+n*4,filters[n]);}std::memcpy(p+64,constants[0].data(),96);')
s=s.replace('n*432','n*192').replace('vertices[begin+n].data(),48','vertices[begin+n].data(),64').replace('n*384','n*128').replace('std::memcpy(record+48,q+256,128)','std::memcpy(record+64,q+80,40)').replace('sample<16','sample<5').replace('const auto slot=0','const auto slot=sample?1:0').replace('record+176+sample*16','record+104+sample*16')
p.write_text(s)
p=d/'compare.cpp';s=p.read_text().replace('invalid[1].width=1','invalid[2].width=1');p.write_text(s)
# Record the full reproducible source set; CPU reference remains test-only.
manifest={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in d.iterdir() if p.is_file()};(r/'ptl-complex-integration-v1-source.json').write_text(json.dumps(manifest,indent=2)+'\n')
with tarfile.open(r/'ptl-complex-integration-v1.tar.gz','w:gz') as archive:archive.add(d,arcname=d.name)
s=(r/'build-ptl-long-shader-integration-v4.ps1').read_text().replace('ptl-long-shader-integration-v4','ptl-complex-integration-v1').replace('nr_ptl_four_tap_compare','nr_ptl_complex_compare');(r/'build-ptl-complex-integration-v1.ps1').write_text(s)
print(json.dumps(dict(prepared=True,model_sha256=args.model_sha256,shader_words=len(found[0]),source_files=len(manifest))))
