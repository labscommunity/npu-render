  aggregate=by_shader.setdefault(x['pixel_sha256'],dict(requests=0,overlap_ms=0,whole_request_provider_calls=0,whole_request_backend_compile_ms=0));aggregate['requests']+=1;aggregate['overlap_ms']+=x['overlap_ms'];aggregate['whole_request_provider_calls']+=x['provider_calls'];aggregate['whole_request_backend_compile_ms']+=x['backend_compile_ms']

  c=child_times[ticket];assert p['pid']==c['client_pid'] and p['sequence']==c['sequence']

 if publication:result['input_to_publication_ms']=(publication['qpc']-start)*1000/frequency;result['publication_to_observation_ms']=(end-publication['qpc'])*1000/frequency

print(json.dumps({k:[{key:value for key,value in event.items() if key not in ('overlapping_draws','first_matching_publication','first_matching_submission')} for event in v] if k=='controlled_events' else v for k,v in report.items() if k not in ('source_sha256',)},indent=2))
