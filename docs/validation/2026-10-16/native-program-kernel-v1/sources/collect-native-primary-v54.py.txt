from pathlib import Path
import base64,subprocess,json,hashlib,concurrent.futures
r=Path("results/render-driver")
directories=['ptl-six-span-program-build-v1','ptl-six-span-program-build-v2','ptl-six-span-program-v3']
extras=['ptl-six-span-program-v1-source.json','ptl-six-span-program-v2-source.json','ptl-six-span-program-v3-source.json']
source_files=['ptl-six-span-program-v1.c','ptl-six-span-program-v2.c','ptl_program_fixtures_v2.py','ptl_geometry_fixtures_v2.py','qualify-ptl-six-span-program-v1.py','qualify-ptl-six-span-program-v2.py','qualify-ptl-six-span-program-v3.py','run-ptl-program-v1.ps1','run-ptl-program-v2.ps1','run-ptl-program-v3.ps1']
ps=r'''$ErrorActionPreference='Stop';$ProgressPreference='SilentlyContinue';$r="$env:LOCALAPPDATA\npu-render"
foreach($trial in @('ptl-six-span-program-v3')){$q=Get-Content "$r\vm\$trial\report.json" -Raw|ConvertFrom-Json;if(!$q.completed){throw 'Cannot archive active comparison'}}
function ArtifactHash([string]$path){$stream=[IO.File]::Open($path,[IO.FileMode]::Open,[IO.FileAccess]::Read,([IO.FileShare]::ReadWrite -bor [IO.FileShare]::Delete));try{return (Get-FileHash -InputStream $stream -Algorithm SHA256).Hash}finally{$stream.Dispose()}}
$items=@(foreach($name in @('''+','.join("'"+d+"'" for d in directories)+r''')){
 Get-ChildItem "$r\vm\$name" -File -Recurse|Where-Object {($_.Extension -in @('.json','.jsonl','.log','.txt','.inf','.ps1','.py','.c','.cpp','.hpp')) -and ($_.Name -notlike 'client-*' -or ($_.Name -like '*.stderr.log' -and $_.Length -gt 0))}|ForEach-Object {
 if($_.Length -gt 18000000){throw 'Unexpected artifact size'}
 [ordered]@{source=$_.FullName.Substring(("$r\vm\").Length).Replace('\','/');bytes=$_.Length;sha256=(ArtifactHash $_.FullName)}
 }
}
foreach($name in @('''+','.join("'"+d+"'" for d in extras+source_files)+r''')){[ordered]@{source=$name;bytes=(Get-Item "$r\vm\$name").Length;sha256=(ArtifactHash "$r\vm\$name")}}

);$items|ConvertTo-Json -Depth 5
'''
p=r/'hash-native-primary-v54.ps1';p.write_text(ps)
subprocess.run(['scp','-q',str(p),'delta:AppData/Local/npu-render/vm/'],capture_output=True,check=True,timeout=30)
invoke=r'''$r="$env:LOCALAPPDATA\npu-render";& ([scriptblock]::Create([IO.File]::ReadAllText("$r\vm\hash-native-primary-v54.ps1")))'''
result=subprocess.run(['ssh','-o','BatchMode=yes','-o','ConnectTimeout=8','delta','powershell.exe -NoProfile -NonInteractive -EncodedCommand '+base64.b64encode(invoke.encode('utf-16-le')).decode()],capture_output=True,timeout=45)
if result.returncode:print(result.stderr.decode(errors='replace')[:2000]);raise SystemExit(result.returncode)
receipts=json.loads(result.stdout.decode('utf-8-sig'));assert 0<len(receipts)<1000
(r/'native-primary-v54-artifact-sha256.json').write_text(json.dumps(receipts,indent=2)+'\n')
def fetch(item):
    name=item['source'];assert '..' not in name and not name.startswith('/');dest=r/'primary-raw-v54'/name;dest.parent.mkdir(parents=True,exist_ok=True)
    subprocess.run(['scp','-q','-o','BatchMode=yes','delta:AppData/Local/npu-render/'+item.get('remote_source','vm/'+name),str(dest)],capture_output=True,check=True,timeout=90)
    data=dest.read_bytes();assert len(data)==item['bytes'] and hashlib.sha256(data).hexdigest()==item['sha256'].lower()
    if name in source_files:assert data==(r/name).read_bytes()
    return len(data)
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:sizes=list(pool.map(fetch,receipts))
print(json.dumps(dict(hash_verified_files=len(sizes),bytes=sum(sizes),source_files_matched_local=len(source_files))))
