cmake_minimum_required(VERSION 3.25)
project(npu_repeated_input LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The probe requires x64 Windows/MSVC")
endif()
add_executable(nr_repeated_input repeated_input.cpp)
target_compile_features(nr_repeated_input PRIVATE cxx_std_17)
target_compile_definitions(nr_repeated_input PRIVATE UNICODE _UNICODE)
target_compile_options(nr_repeated_input PRIVATE /W4 /WX)
target_link_libraries(nr_repeated_input PRIVATE user32 gdi32 wtsapi32 advapi32)
set_property(TARGET nr_repeated_input PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
