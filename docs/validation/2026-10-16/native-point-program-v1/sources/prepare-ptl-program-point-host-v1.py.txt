from pathlib import Path

for name in ['reference_triangle.hpp','reference_triangle.cpp','long_shader_reference.hpp']:
 shutil.copyfile(r/'ptl-long-shader-integration-v4'/name,d/name)

 if p.suffix in ('.txt','.cpp'):
  s=p.read_text().replace('nr_ptl_four_tap_compare','nr_ptl_program_point_compare');p.write_text(s)
