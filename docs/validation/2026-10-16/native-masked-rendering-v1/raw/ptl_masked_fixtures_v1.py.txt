"""Independent FP32 oracle for the bounded masked-texture component.

Coverage/interpolation comes from the previously independent integer-edge
oracle. Texture addressing, scalar channel arithmetic and output packing are
computed here independently of the C implementation. No device calls.
"""
import struct

import numpy as np

from ptl_geometry_fixtures_v2 import fixture as geometry_fixture

SPAN = 262144
F = np.float32


def fixture(index):
    if index >= 114:
        payload, _ = fixture(49)
        payload = bytearray(payload)
        offset, value = [(16, 3), (256+68, 2), (256+72, 1),
                         (256+16, 0x3f800000), (256+20, 0x7fc00000),
                         (256, 0x7f800000)][index-114]
        struct.pack_into('<I', payload, offset, value)
        return bytes(payload), bytes(SPAN)
    seed = index % 96
    phase = int(seed >= 48)
    rng = np.random.default_rng(910000 + seed)
    target = [28, 87][seed % 2]
    mask = 1 + seed % 15
    blend = (seed // 2) % 2
    layouts = [(1, 1), (2, 3), (2048, 8), (1024, 1056), (704, 704), (8192, 1)]
    resources = [(*layouts[seed % 6], [28, 87][seed % 2]),
                 (*layouts[(seed + 2) % 6], [87, 28][(seed // 2) % 2])]
    payload = bytearray(SPAN)
    expected = bytearray(SPAN)
    if not phase:
        geometry, coverage = geometry_fixture((seed * 3) % 96)
        count, vertices, cull, ccw = struct.unpack_from('<4I', geometry, 8)
        payload[256:] = geometry[256:]
        payload[32:48] = geometry[32:48]
        hits = struct.unpack_from('<I', coverage, 8)[0]
        for n in range(count):
            base = 16 + n * 80
            if not struct.unpack_from('<I', coverage, base)[0]:
                continue
            struct.pack_into('<I', expected, base, 1)
            expected[base+16:base+32] = coverage[base+16:base+32]
            for sample, (width, height, _) in enumerate(resources):
                uv = struct.unpack_from('<2f', coverage, base + 32 + sample * 16)
                xy = []
                for value, size in zip(uv, (width, height)):
                    clamped = F(min(1, max(0, value)))
                    scaled = F(F(clamped * F(size)) - F(.5))
                    xy.append(F(min(size - 1, max(0, scaled))))
                x0, y0 = int(xy[0]), int(xy[1])
                x1, y1 = min(width-1, x0+1), min(height-1, y0+1)
                struct.pack_into('<4I', expected, base + 32 + sample * 16,
                                 y0*width+x0, y0*width+x1, y1*width+x0, y1*width+x1)
                struct.pack_into('<2f', expected, base + 64 + sample * 8,
                                 F(xy[0]-F(x0)), F(xy[1]-F(y0)))
    else:
        count = [0, 1, 6, 31, 127, 511, 1024, 3072][seed % 8]
        vertices = cull = ccw = hits = 0
        for n in range(count):
            p = 256+n*80
            covered = int(n % 7 != 1)
            color = rng.integers(-64, 769, 4).astype(np.float32) / F(256)
            fractions = rng.integers(0, 4096, 4).astype(np.float32) / F(4096)
            texels = rng.integers(0, 2**32, (2, 4), dtype=np.uint32)
            background = int(rng.integers(0, 2**32, dtype=np.uint32))
            if n % 11 == 0:
                # Exact half-byte ties and zero/unit alpha.
                color = np.array([.5/255, 1.5/255, 2.5/255, (n//11) % 2], np.float32)
                texels.fill(0xffffffff)
            struct.pack_into('<8f8I4I', payload, p, *color, *fractions,
                             *map(int, texels.reshape(-1)), background, covered, 0, 0)
            result = background
            if covered:
                hits += 1
                samples = []
                for sample in range(2):
                    tx, ty = fractions[sample*2:sample*2+2]
                    channels = []
                    for channel in range(4):
                        byte = 2-channel if resources[sample][2] == 87 and channel < 3 else channel
                        values = [F(F((int(v) >> (8*byte)) & 255) / F(255)) for v in texels[sample]]
                        upper = F(F(values[0]*F(F(1)-tx)) + F(values[1]*tx))
                        lower = F(F(values[2]*F(F(1)-tx)) + F(values[3]*tx))
                        channels.append(F(F(upper*F(F(1)-ty)) + F(lower*ty)))
                    samples.append(channels)
                shaded = [F(min(1, max(0, F(F(color[c]*samples[0][c])*samples[1][3])))) for c in range(4)]
                for byte in range(4):
                    channel = 2-byte if target == 87 and byte < 3 else byte
                    if not mask & (1 << channel):
                        continue
                    value = F(shaded[channel]*F(255))
                    if blend:
                        value = F(value+F(F((background >> (8*byte)) & 255)*F(F(1)-shaded[3])))
                    packed = int(np.rint(F(min(255, max(0, value)))))
                    result = (result & ~(255 << (8*byte))) | (packed << (8*byte))
            struct.pack_into('<I', expected, 16+n*4, result)
    struct.pack_into('<8I', payload, 0, 0x4e524d31, 1, phase, count, vertices, cull, ccw, SPAN)
    struct.pack_into('<4I', payload, 48, target, mask, blend, 0)
    for sample, resource in enumerate(resources):
        struct.pack_into('<3I', payload, 64+sample*12, *resource)
    struct.pack_into('<4I', expected, 0, 0x4e524d4f, phase, count, hits)
    if index >= 96:
        offset, value = [(0, 0), (4, 2), (8, 2), (12, 3073), (16, 2049),
                         (20, 3), (24, 2), (28, SPAN-1), (48, 0), (52, 0),
                         (56, 2), (60, 1), (64, 0), (68, 8193), (72, 0),
                         (80, 0), (84, 0), (256+20, 1)][index-96]
        struct.pack_into('<I', payload, offset, value)
        expected = bytearray(SPAN)
    return bytes(payload), bytes(expected)
