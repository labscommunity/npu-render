cmake_minimum_required(VERSION 3.25)
project(nr_physical_gpu_control LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Windows/MSVC required")
endif()
add_executable(nr_physical_gpu_probe probe.cpp)
target_compile_features(nr_physical_gpu_probe PRIVATE cxx_std_17)
target_compile_options(nr_physical_gpu_probe PRIVATE /W4 /WX)
target_link_libraries(nr_physical_gpu_probe PRIVATE dxgi d3d11)
set_property(TARGET nr_physical_gpu_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
