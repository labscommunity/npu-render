def rows(p):return [json.loads(x) for x in p.read_text(encoding='utf-8-sig').splitlines() if x.startswith('{') and x.endswith('}')]
paths=[vm/'ptl-native-worker-v1-full/replay',vm/'ptl-native-worker-v2-retry-full/replay'];proof=[json.loads((p/'report.json').read_text()) for p in paths];assert all(p['passed'] and p['delivery_acknowledgments']==972 for p in proof)
