cmake_minimum_required(VERSION 3.25)
project(cpu_precision_check LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
foreach(name span_blend packed_blend)
 add_executable(nr_${name}_test "${NR_SOURCE_ROOT}/native/tests/${name}_test.cpp")
 target_compile_features(nr_${name}_test PRIVATE cxx_std_17)
 target_link_libraries(nr_${name}_test PRIVATE openvino::runtime)
endforeach()
enable_testing()
add_test(NAME span_blend_cpu COMMAND nr_span_blend_test CPU)
add_test(NAME packed_blend_cpu COMMAND nr_packed_blend_test)
