import json,tempfile
from pathlib import Path
import numpy as np
import openvino as ov
from openvino import opset13 as o
n=256
colors=o.parameter([n,4],np.float32,name='colors');bg=o.parameter([n,4],np.uint8,name='background');mask=o.parameter([n,1],np.int32,name='coverage')
def c(x):return o.constant(x,np.float32)
def ci(x):return o.constant(x,np.int32)
def choose(cond,a,b):
 shape=[n,4];a=o.broadcast(a,shape);b=o.broadcast(b,shape);cond=o.broadcast(cond,shape)
 index=o.add(o.constant(np.arange(n*4,dtype=np.int32)),o.multiply(o.reshape(o.convert(o.logical_not(cond),np.int32),[n*4],False),ci(n*4)))
 return o.reshape(o.gather(o.concat([o.reshape(a,[n*4],False),o.reshape(b,[n*4],False)],0),index,0),shape,False)
fg=o.gather(colors,[2,1,0,3],1);alpha=o.gather(colors,[3],1)
blended=o.add(o.multiply(fg,c(255)),o.multiply(o.convert(bg,np.float32),o.subtract(c(1),alpha)))
value=choose(o.less(blended,c(0)),c(0),choose(o.greater(blended,c(255)),c(255),blended))
base=o.bitwise_and(o.convert(value,np.int32),ci(255));fraction=o.subtract(value,o.convert(base,np.float32))
increment=o.bitwise_or(o.convert(o.greater(fraction,c(.5)),np.int32),o.bitwise_and(o.convert(o.equal(fraction,c(.5)),np.int32),o.bitwise_and(base,ci(1))))
quantized=o.add(base,increment)
output=o.convert(o.add(o.multiply(quantized,mask),o.multiply(o.convert(bg,np.int32),o.subtract(ci(1),mask))),np.uint8)
with tempfile.TemporaryDirectory(ignore_cleanup_errors=True) as d:
 m=ov.Model([output,fg,blended,value,base,fraction,increment,quantized],[colors,bg,mask]);path=Path(d)/'m.xml'
 for node in m.get_ops():node.set_rt_info('','precise_0')
 ov.serialize(m,path);core=ov.Core();compiled=core.compile_model(core.read_model(path),'NPU',{'EXECUTION_MODE_HINT':'ACCURACY'})
 rng=np.random.default_rng(97814)
 for frame in range(4):
  cc=rng.uniform(-.125,1.125,(n,4)).astype(np.float32);cc[:,3]=rng.uniform(0,1,n).astype(np.float32)
  bb=rng.integers(0,256,(n,4),dtype=np.uint8);mm=rng.integers(0,2,(n,1),dtype=np.int32)
  if frame==0:cc[:]=np.array([0,0,0,1],np.float32);mm[:]=1
  if frame==1:cc[:]=np.array([0,0,0,1],np.float32);mm[:]=0
  if frame==2:cc[:,3]=1
  ff=cc[:,[2,1,0,3]];expected=np.rint(np.clip(ff*255+bb.astype(np.float32)*(np.float32(1)-cc[:,3:4]),0,255)).astype(np.uint8)
  expected=np.where(mm.astype(bool),expected,bb)
  outputs=compiled({'colors':cc,'background':bb,'coverage':mm});actual=outputs[0]
  print(json.dumps({'intermediates':[outputs[i][:2].tolist() for i in range(1,8)]}),flush=True)
  error=np.abs(actual.astype(np.int16)-expected.astype(np.int16))
  print(json.dumps({'frame':frame,'execution_device':compiled.get_property('EXECUTION_DEVICES'),'exact':bool(np.array_equal(actual,expected)),'max_byte_error':int(error.max()),'masked_background_exact':bool(np.all(actual[mm[:,0]==0]==bb[mm[:,0]==0])),'desktop_offload_tested':False}),flush=True)
  if not np.array_equal(actual,expected):print(json.dumps({'actual4':actual[:4].tolist(),'expected4':expected[:4].tolist()}),flush=True)
