import json,os
from pathlib import Path
import numpy as np
import openvino as ov
from openvino import opset13 as op
x=op.parameter([16,1],np.float32,name='x');const=lambda v:op.constant(np.float32(v));integer=lambda v:op.constant(np.int32(v))
def choose(c,a,b):
 c=op.broadcast(c,[16,1]);a=op.broadcast(a,[16,1]);b=op.broadcast(b,[16,1]);index=op.add(op.constant(np.arange(16,dtype=np.int32)),op.multiply(op.reshape(op.convert(op.logical_not(c),'i32'),[16],False),integer(16)))
 return op.gather(op.concat([a,b],0),index,integer(0))
mag=op.abs(x);original=op.less(mag,const(8388608));small=op.less(op.subtract(mag,const(8388608)),const(0));safe=choose(small,mag,const(0));whole=op.bitwise_and(op.convert(safe,'i32'),integer(0x7fffffff));back=op.convert(whole,'f32');frac=op.subtract(safe,back)
negative=choose(op.equal(frac,const(0)),const(0),op.subtract(const(1),frac));finite=choose(op.less(x,const(0)),negative,frac);finite=choose(op.less(mag,const(np.finfo(np.float32).tiny)),const(0),finite);invalid=op.logical_or(op.is_nan(x),op.is_inf(x));final=choose(invalid,const(np.nan),finite)
values=np.array([-17.75,-3,-.125,-0.,0,.125,.75,1,3.125,8388607.5,8388608,np.finfo(np.float32).max,np.finfo(np.float32).smallest_subnormal,-np.finfo(np.float32).smallest_subnormal,np.inf,np.nan],np.float32).reshape(16,1)
model=ov.Model([original,small,safe,whole,frac,final],[x]);core=ov.Core();directory=Path(os.environ['LOCALAPPDATA'])/'npu-render/vm/fraction-v143';directory.mkdir(exist_ok=False)
for node in model.get_ops():node.get_rt_info()['precise_0']=''
ov.serialize(model,str(directory/'model.xml'),str(directory/'model.bin'));model=core.read_model(str(directory/'model.xml'))
for backend in ['CPU','NPU']:
 compiled=core.compile_model(model,backend,{'EXECUTION_MODE_HINT':'ACCURACY'} if backend=='NPU' else {'INFERENCE_PRECISION_HINT':'f32'})
 result=compiled([values]);print(json.dumps({'backend':backend,'execution_devices':compiled.get_property('EXECUTION_DEVICES'),'rows':[{'input':float(values[n,0]),'stages':[float(result[i][n,0]) for i in range(6)]} for n in range(16)]}),flush=True)
