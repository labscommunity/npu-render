cmake_minimum_required(VERSION 3.25)
project(solid_diagnostic LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Registered source")
add_executable(solid_diagnostic solid_rectangle.cpp solid_diagnostic.cpp)
target_include_directories(solid_diagnostic PRIVATE "${NR_RENDER_SOURCE}")
target_link_libraries(solid_diagnostic PRIVATE openvino::runtime)
target_compile_features(solid_diagnostic PRIVATE cxx_std_17)
target_compile_options(solid_diagnostic PRIVATE /W4 /WX)
set_property(TARGET solid_diagnostic PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
