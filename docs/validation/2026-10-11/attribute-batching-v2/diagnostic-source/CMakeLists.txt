cmake_minimum_required(VERSION 3.25)
project(attribute_diagnostic LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Registered source")
set(NR_RENDER_LIBRARY "" CACHE FILEPATH "Qualified renderer library")
set(NR_CACHE_LIBRARY "" CACHE FILEPATH "Provider cache library")
add_executable(attribute_test triangle_texture_draw.cpp triangle_texture_draw_test.cpp)
add_executable(attribute_replay triangle_texture_draw.cpp "${NR_RENDER_SOURCE}/draw_replay_probe.cpp")
foreach(target attribute_test attribute_replay)
 target_include_directories(${target} PRIVATE "${NR_RENDER_SOURCE}")
 target_link_libraries(${target} PRIVATE "${NR_RENDER_LIBRARY}" "${NR_CACHE_LIBRARY}" openvino::runtime d3dcompiler bcrypt)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE /W4 /WX)
 set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
