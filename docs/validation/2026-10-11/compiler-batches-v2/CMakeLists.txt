cmake_minimum_required(VERSION 3.25)
project(npu_render_compiler_batch_probe LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Expected Windows x64 MSVC")
endif()
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Qualified registered renderer source directory")
add_executable(nr_compositor_batch_probe compositor_batch_probe.cpp vertex_executor.cpp)
target_include_directories(nr_compositor_batch_probe PRIVATE "${NR_RENDER_SOURCE}")
target_link_libraries(nr_compositor_batch_probe PRIVATE openvino::runtime)
target_compile_features(nr_compositor_batch_probe PRIVATE cxx_std_17)
target_compile_options(nr_compositor_batch_probe PRIVATE /W4 /WX)
set_property(TARGET nr_compositor_batch_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
