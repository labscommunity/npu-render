cmake_minimum_required(VERSION 3.25)
project(crop_admission_diagnostic LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Registered source")
set(NR_RENDER_LIBRARY "" CACHE FILEPATH "Qualified renderer library")
set(NR_CACHE_LIBRARY "" CACHE FILEPATH "Provider cache library")
add_executable(nr_crop_admission_probe draw_replay_probe.cpp)
target_include_directories(nr_crop_admission_probe PRIVATE "${NR_RENDER_SOURCE}")
target_link_libraries(nr_crop_admission_probe PRIVATE "${NR_RENDER_LIBRARY}" "${NR_CACHE_LIBRARY}" openvino::runtime bcrypt)
target_compile_features(nr_crop_admission_probe PRIVATE cxx_std_17)
target_compile_options(nr_crop_admission_probe PRIVATE /W4 /WX)
set_property(TARGET nr_crop_admission_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
