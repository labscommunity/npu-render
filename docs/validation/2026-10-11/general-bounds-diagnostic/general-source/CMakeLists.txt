cmake_minimum_required(VERSION 3.25)
project(bounds_diagnostic LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 required")
endif()
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
set(NR_RENDER_SOURCE "" CACHE PATH "Registered source")
add_library(bounds_vertex STATIC vertex_executor.cpp)
target_include_directories(bounds_vertex PUBLIC "${NR_RENDER_SOURCE}")
target_link_libraries(bounds_vertex PUBLIC openvino::runtime d3dcompiler)
add_executable(bounds_vertex_test "${NR_RENDER_SOURCE}/vertex_test.cpp")
add_executable(bounds_texture_test "${NR_RENDER_SOURCE}/texture_test.cpp")
add_executable(bounds_loop_test loop_diagnostic.cpp)
foreach(target bounds_vertex bounds_vertex_test bounds_texture_test bounds_loop_test)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE /W4 /WX)
 set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
foreach(target bounds_vertex_test bounds_texture_test bounds_loop_test)
 target_link_libraries(${target} PRIVATE bounds_vertex)
endforeach()
