pattern = re.compile("|".join(re.escape(x) for x in sorted(changes, key=len, reverse=True)))


def rename(text):
    return pattern.sub(lambda match: changes[match[0]], text)


files = [n for n in json.loads((r / "ptl-combined-live-v15-upload.json").read_text())
         if not n.endswith("-source.json")]
for name in files:
    s = rename((r / name).read_text())
    if name == "native_shared_app_v19.py":
        # This one reference is the active registration, not a prior cleanup gate.
        assert s.count("native-registration-v52/qualification/report.json") == 1
        s = s.replace("native-registration-v52/qualification/report.json",
                      "native-registration-v53/qualification/report.json")
    if name == "run-native-live-v63.ps1":
        # The established generation audit checks singleton shared providers.
        # Solid/four-tap/texture providers may legitimately import more than once.
        needle = "  if(@($routes|Sort-Object -Unique).Count -ne $routes.Count){throw 'Duplicate turbo import within child generation'}"
        assert s.count(needle) == 1
        s = s.replace(needle, "  if(@($routes|Where-Object {$_ -notin @('program','solid','geometry','four_tap','point_rectangle','complex_shader','resident_clip','texture_shader','long_shader','masked')}).Count){throw 'Unknown turbo import route'}")
    if name.endswith(".py"):
        ast.parse(s)
    dest = r / rename(name)
    if dest == r / name:
        assert dest.read_text() == s
    else:
        assert not dest.exists(), dest
        dest.write_text(s)
for old in ("native-registration-v52-source.json", "native-desktop-v73-source.json"):
    names = [rename(n) for n in json.loads((r / old).read_text())]
    (r / rename(old)).write_text(json.dumps({
        n: hashlib.sha256((r / n).read_bytes()).hexdigest() for n in names
    }, indent=2) + "\n")

(r / "ptl-combined-live-v16-upload.json").write_text(json.dumps(uploads, indent=2) + "\n")
manifest = {n: hashlib.sha256((r / n).read_bytes()).hexdigest() for n in uploads}
p = r / "ptl-turbo-live-v2-source.json"
