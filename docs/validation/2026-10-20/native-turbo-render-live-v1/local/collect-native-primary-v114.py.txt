"""Archive completed turbo renderer trials, compressing large evidence losslessly."""
from pathlib import Path
import base64
import concurrent.futures
import gzip
import hashlib
import json
import subprocess
import tempfile

r = Path("results/render-driver")
d = Path("docs/validation/2026-10-20/native-turbo-render-live-v1")
directories = ['ptl-combined-desktop-v15', 'ptl-combined-desktop-v16', 'native-registration-v52', 'native-registration-v53', 'ptl-native-live-v27', 'ptl-native-live-v28', 'native-live-v63', 'native-live-v64', 'native-desktop-admission-v73', 'native-desktop-admission-v74', 'native-topology-v126', 'native-topology-v127', 'native-topology-v128', 'native-topology-v129', 'native-topology-v130', 'native-topology-v131', 'native-topology-primary-v49', 'native-topology-primary-v50', 'native-latency-v28-analysis', 'native-presentation-v25-analysis', 'native-repeated-input-v11-analysis']
source_files = ['run-native-registration-v52.ps1', 'recover-native-registration-v52.ps1', 'qualify-native-registered-v52.py', 'native_shared_app_v19.py', 'native_desktop_bootstrap_v47.py', 'run-native-desktop-v73.ps1', 'recover-native-desktop-v73.ps1', 'run-native-live-v63.ps1', 'start-native-topology-v126.ps1', 'start-native-topology-v128.ps1', 'dismiss-native-saver-v38.ps1', 'run-ptl-native-live-v27.ps1', 'analyze-native-latency-v25.py', 'analyze-native-presentation-v24.py', 'run-ptl-combined-desktop-v15.ps1', 'start-native-topology-v127.ps1', 'native-registration-v52-source.json', 'native-desktop-v73-source.json', 'native_shared_correlation_v2.py', 'native_provider_generation_audit_v1.py', 'run-native-registration-v53.ps1', 'recover-native-registration-v53.ps1', 'qualify-native-registered-v53.py', 'native_shared_app_v20.py', 'native_desktop_bootstrap_v48.py', 'run-native-desktop-v74.ps1', 'recover-native-desktop-v74.ps1', 'run-native-live-v64.ps1', 'start-native-topology-v129.ps1', 'start-native-topology-v131.ps1', 'dismiss-native-saver-v39.ps1', 'run-ptl-native-live-v28.ps1', 'analyze-native-latency-v26.py', 'analyze-native-presentation-v25.py', 'run-ptl-combined-desktop-v16.ps1', 'start-native-topology-v130.ps1', 'native-registration-v53-source.json', 'native-desktop-v74-source.json', 'ptl-turbo-live-v1-source.json', 'ptl-turbo-live-v2-source.json']
ps = r"""$ErrorActionPreference='Stop';$ProgressPreference='SilentlyContinue';$r="$env:LOCALAPPDATA\npu-render"
foreach($trial in @('ptl-combined-desktop-v16')){
 $q=Get-Content "$r\vm\$trial\report.json" -Raw|ConvertFrom-Json;if(!$q.completed){throw 'Cannot archive active experiment'}
}
function Item([string]$name,[string]$path,[string]$remote){
 if(!(Test-Path $path)){throw "Missing evidence: $name"}
 $stream=[IO.File]::Open($path,[IO.FileMode]::Open,[IO.FileAccess]::Read,([IO.FileShare]::ReadWrite -bor [IO.FileShare]::Delete))
 try{$h=(Get-FileHash -InputStream $stream).Hash}finally{$stream.Dispose()}
 [ordered]@{source=$name;remote_source=$remote;bytes=(Get-Item $path).Length;sha256=$h}
}
$items=@(
foreach($name in @(""" + ",".join("'" + n + "'" for n in directories) + r""")){
 if(!(Test-Path "$r\vm\$name")){continue}
 Get-ChildItem "$r\vm\$name" -File -Recurse|Where-Object {
  ($_.Extension -in @('.json','.jsonl','.log','.txt','.inf','.ps1','.py','.c','.cpp','.hpp')) -and
  ($_.Name -notlike 'client-*' -or ($_.Name -like '*.stderr.log' -and $_.Length -gt 0))
 }|ForEach-Object {
  if($_.Length -gt 30000000){throw 'Unexpected artifact size'}
  $n=$_.FullName.Substring(("$r\vm\").Length).Replace('\','/');Item $n $_.FullName ("vm/"+$n)
 }
}
foreach($name in @(""" + ",".join("'" + n + "'" for n in source_files) + r""")){
 Item $name "$r\vm\$name" ("vm/"+$name)
}
);$items|ConvertTo-Json -Depth 5
"""
(r / "hash-native-primary-v114.ps1").write_text(ps)
command = "powershell.exe -NoProfile -NonInteractive -EncodedCommand " + base64.b64encode(ps.encode("utf-16-le")).decode()
result = subprocess.run(["ssh", "-o", "BatchMode=yes", "-o", "ConnectTimeout=8", "delta", command],
                        capture_output=True, timeout=45)
if result.returncode:
    print(result.stderr.decode(errors="replace")[:2000])
    raise SystemExit(result.returncode)
receipts = json.loads(result.stdout.decode("utf-8-sig"))
assert 0 < len(receipts) < 1500
assert len({x["source"] for x in receipts}) == len(receipts)
(r / "native-primary-v114-artifact-sha256.json").write_text(json.dumps(receipts, indent=2) + "\n")
d.mkdir(exist_ok=True)


def fetch(item):
    name = item["source"]
    assert ".." not in name and not name.startswith("/")
    relative = "raw/" + name + (".txt" if name.endswith(".py") else "")
    compressed = item["bytes"] >= 65536
    if compressed:
        relative += ".gz"
    dest = d / relative
    if dest.exists():
        data = gzip.decompress(dest.read_bytes()) if compressed else dest.read_bytes()
    else:
        with tempfile.TemporaryDirectory(dir=r) as tmp:
            downloaded = Path(tmp) / "evidence"
            subprocess.run(["scp", "-q", "-o", "BatchMode=yes",
                            "delta:AppData/Local/npu-render/" + item["remote_source"], str(downloaded)],
                           capture_output=True, check=True, timeout=90)
            data = downloaded.read_bytes()
        assert len(data) == item["bytes"] and hashlib.sha256(data).hexdigest() == item["sha256"].lower()
        if name in source_files and (r / name).exists():
            assert data == (r / name).read_bytes()
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_bytes(gzip.compress(data, mtime=0) if compressed else data)
    assert len(data) == item["bytes"] and hashlib.sha256(data).hexdigest() == item["sha256"].lower()
    stored = dest.read_bytes()
    return dict(path=relative, bytes=len(stored), sha256=hashlib.sha256(stored).hexdigest(),
                remote_source=item["remote_source"], source_bytes=len(data),
                source_sha256=item["sha256"].lower(), compression="gzip" if compressed else None)


with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    items = list(pool.map(fetch, receipts))
for name in ('collect-native-primary-v114.py','hash-native-primary-v114.ps1','prepare-turbo-render-live-v1.py','prepare-turbo-render-live-v2.py'):
    data = (r / name).read_bytes()
    dest = d / "local" / (name + (".txt" if name.endswith(".py") else ""))
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
    items.append(dict(path=str(dest.relative_to(d)), bytes=len(data),
                      sha256=hashlib.sha256(data).hexdigest(), local_source=str(r / name)))
(d / "artifact-sha256.json").write_text(json.dumps(items, indent=2) + "\n")
(d / ".gitattributes").write_text("raw/** -text whitespace=cr-at-eol,-blank-at-eol,-blank-at-eof\nlocal/** -text whitespace=cr-at-eol,-blank-at-eol,-blank-at-eof\n*.gz binary\n")
print(json.dumps(dict(artifacts=len(items), stored_bytes=sum(x["bytes"] for x in items),
                      remote_bytes=sum(x["bytes"] for x in receipts),
                      remote_sha256_verified=True, lossless_compression=True)))
