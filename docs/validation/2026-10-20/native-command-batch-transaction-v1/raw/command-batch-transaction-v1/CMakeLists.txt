cmake_minimum_required(VERSION 3.25)
project(nr_command_batch_transaction LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
add_executable(nr_command_batch_transaction probe.cpp)
target_compile_features(nr_command_batch_transaction PRIVATE cxx_std_17)
target_compile_options(nr_command_batch_transaction PRIVATE /W4 /WX)
target_compile_definitions(nr_command_batch_transaction PRIVATE NOMINMAX)
target_link_libraries(nr_command_batch_transaction PRIVATE openvino::runtime bcrypt)
set_property(TARGET nr_command_batch_transaction PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
