"""Separate publication, presentation statistics, and completed pixel reads."""
from pathlib import Path
import argparse
import hashlib
import json
from presentation_counter_match_v1 import match_display_observation
from pixel_observation_match_v1 import match_owned_pixel_observation

p = argparse.ArgumentParser()
p.add_argument('--root', type=Path, required=True)
p.add_argument('--timing', type=Path)
p.add_argument('--output', type=Path, required=True)
a = p.parse_args()
if a.timing is None:
    a.timing = a.root / 'native-latency-v23-analysis/report.json'
timing = json.loads(a.timing.read_text(encoding='utf-8-sig'))
present_path = a.root / 'native-live-v59/present/probe.log'
screen_path = a.root / 'native-live-v59/screen/probe.log'
def rows(path):
    return [json.loads(line) for line in path.read_text(encoding='utf-8-sig').splitlines()
            if line.startswith('{') and line.endswith('}')]
present, screen = rows(present_path), rows(screen_path)
frequency = timing['qpc_frequency']
events = []
for event in timing['controlled_events']:
    start = event['input_qpc']
    publication = event['first_matching_publication']
    submission = event['first_matching_submission']
    assert publication and submission and submission['backbuffer_exact']
    assert submission['source_qpc'] == publication['qpc']
    assert start <= publication['qpc'] <= submission['submitted_qpc'] <= submission['accepted_qpc']
    sequence = publication['sequence']
    submitted_count,display=match_display_observation(present,sequence,submission)
    read = match_owned_pixel_observation(screen,start,lambda point: point.get('known_' + event['color']))
    assert read['qpc'] <= read['read_completed_qpc']
    update_qpc = read['last_present_qpc']
    assert 0 <= update_qpc <= read['read_completed_qpc']
    if update_qpc:
        assert update_qpc >= start
    else:
        assert read['accumulated_frames'] == 0  # Documented pointer-only update.
    by_shader = {}
    for draw in event['overlapping_draws']:
        elapsed = max(0, min(publication['qpc'], draw['finished_qpc']) - max(start, draw['entered_qpc'])) * 1000 / frequency
        if elapsed:
            entry = by_shader.setdefault(draw['pixel_sha256'], {'overlap_ms': 0, 'requests': 0})
            entry['overlap_ms'] += elapsed
            entry['requests'] += 1
    busy = sum(x['overlap_ms'] for x in by_shader.values())
    before = (publication['qpc'] - start) * 1000 / frequency
    assert busy <= before
    events.append(dict(event=event['event'], source_sequence=sequence, submitted_present_count=submitted_count, display_matched_by_submitted_counter=True,
        input_to_publication_ms=before,
        input_to_present_acceptance_ms=(submission['accepted_qpc']-start)*1000/frequency,
        input_to_matching_display_statistics_ms=(display['sync_qpc']-start)*1000/frequency,
        publication_to_matching_display_statistics_ms=(display['sync_qpc']-publication['qpc'])*1000/frequency,
        input_to_observer_acquisition_started_ms=(read['qpc']-start)*1000/frequency,
        input_to_pixel_read_completed_ms=(read['read_completed_qpc']-start)*1000/frequency,
        input_to_observed_desktop_update_ms=(update_qpc-start)*1000/frequency if update_qpc else None,
        pixel_read_has_desktop_update_timestamp=bool(update_qpc),
        pixel_read_pointer_update_only=not bool(update_qpc),
        serialized_child_before_publication_ms=busy,
        other_before_publication_ms=before-busy, by_shader_before_publication=by_shader,
        matching_display_statistics=display, first_matching_pixel_read=read))
report = dict(completed=True, passed=True, active_snapshot=timing['active_snapshot'],
    events=events, full_desktop_offload=False, tahoma_tested=False,
    source_sha256={str(path.relative_to(a.root)) if path.is_relative_to(a.root) else path.name:
        hashlib.sha256(path.read_bytes()).hexdigest() for path in [a.timing, present_path, screen_path]},
    limitations='Two controlled events, no matched A/B or p95. Presentation statistics and independent pixel reads are distinct observations; neither measures panel photons. Observer acquisition can begin before input while AcquireNextFrame waits. Acquisition begin is not read completion. Child overlap after publication is excluded from attribution to producing that frame. Display statistics do not establish that the independent observer saw the same surface update. A pointer-only duplication update has no desktop-update timestamp; its owned pixel read completion remains a separate upper bound and is never substituted for display-counter timing.',
    primary_sources=['https://learn.microsoft.com/en-us/windows/win32/api/dxgi/nf-dxgi-idxgiswapchain-getlastpresentcount',
                     'https://learn.microsoft.com/en-us/windows/win32/api/dxgi/ns-dxgi-dxgi_frame_statistics',
                     'https://learn.microsoft.com/en-us/windows/win32/api/dxgi1_2/ns-dxgi1_2-dxgi_outdupl_frame_info'])
a.output.mkdir(exist_ok=False, parents=True)
(a.output/'report.json').write_text(json.dumps(report, indent=2)+'\n')
print(json.dumps([{k:v for k,v in event.items() if k not in ('by_shader_before_publication', 'matching_display_statistics', 'first_matching_pixel_read')} for event in events]))
