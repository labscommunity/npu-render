"""Independent integer-edge, FP32 sampling and packed-pixel oracle."""
import struct

import numpy as np

from ptl_resident_clip_geometry_v1 import fixture as geometry_fixture

SPAN = 262144
F = np.float32


def fixture(index):
    # All invalid cases alter fields checked before the output loop.
    invalid = [
        (0, 0), (4, 2), (8, 16385), (12, 0), (12, 2049), (12, 4),
        (16, 3), (20, 2), (24, SPAN - 1), (28, 1), (48, 0), (52, 0),
        (56, 2), (60, 1), (64, 0), (68, 8193), (72, 0), (76, 0),
        (84, 0), (88, 2), (92, 2), (96, 0x7f800000), (112, 0x7fc00000),
        (256 + 20, 1),
    ]
    if index >= 120:
        payload, _ = fixture(25)
        payload = bytearray(payload)
        offset, value = invalid[index - 120]
        struct.pack_into("<I", payload, offset, value)
        return bytes(payload), bytes(SPAN)
    seed = index % 96
    geometry, coverage = geometry_fixture((seed * 3) % 96)
    count, vertices, cull, ccw = struct.unpack_from("<4I", geometry, 8)
    rng = np.random.default_rng(1240000 + index)
    layouts = [(1, 1), (2, 3), (8, 8), (32, 32), (64, 64), (128, 1)]
    resources = [
        (*layouts[seed % 6], [28, 87][seed % 2]),
        (*layouts[(seed + 1) % 6], [87, 28][(seed // 2) % 2]),
    ]
    filters = [seed % 2, (seed // 2) % 2]  # 0 linear, 1 point
    target = [28, 87][seed % 2]
    mask = 1 + seed % 15
    blend = (seed // 3) % 2
    bounds = [
        [-1, -1, 1, 1], [0, 0, 1, 1], [.125, .25, .75, 1],
        [-4, -4, 4, 4], [1, 1, -1, -1], [0, 0, 0, 0],
    ]
    clips = [bounds[(seed // 4) % 6], bounds[(seed // 7 + 2) % 6]]
    payload = bytearray(SPAN)
    expected = bytearray(SPAN)
    payload[32:48] = geometry[32:48]
    payload[256:256 + vertices * 96] = geometry[256:256 + vertices * 96]
    textures = []
    end = 256 + vertices * 96
    for width, height, _ in resources:
        texture = rng.integers(0, 2**32, width * height, dtype=np.uint32)
        if index % 11 == 0:
            texture[:] = 0xffffffff
        data = texture.astype("<u4").tobytes()
        payload[end:end + len(data)] = data
        end += len(data)
        textures.append(texture)
    capacity = min(16384, (SPAN - end) // 16)
    assert capacity >= 0
    original_count = count
    count = min(count, capacity)
    if index >= 96 and original_count:
        # Exercise the larger resident capacity by repeating independent tasks.
        count = capacity
    hits = 0
    for n in range(count):
        original = n % original_count
        p = 16 + original * 80
        background = int(rng.integers(0, 2**32, dtype=np.uint32))
        payload[end + n * 16:end + n * 16 + 12] = geometry[196864 + original * 12:196876 + original * 12]
        struct.pack_into("<I", payload, end + n * 16 + 12, background)
        result = background
        covered = struct.unpack_from("<I", coverage, p)[0]
        if covered:
            hits += 1
            alpha = F(struct.unpack_from("<f", coverage, p + 28)[0])
            samples = []
            enabled = []
            for sample, (width, height, fmt) in enumerate(resources):
                x, y, z = struct.unpack_from("<3f", coverage, p + 32 + sample * 16)
                u, v = F(F(x) / F(z)), F(F(y) / F(z))
                assert np.isfinite(u) and np.isfinite(v)
                low_x, low_y, high_x, high_y = clips[sample]
                enabled.append(F(low_x <= u <= high_x and low_y <= v <= high_y))
                half = F(0 if filters[sample] else .5)
                tx = F(min(width - 1, max(0, F(F(F(min(1, max(0, u))) * F(width)) - half))))
                ty = F(min(height - 1, max(0, F(F(F(min(1, max(0, v))) * F(height)) - half))))
                ix, iy = int(tx), int(ty)
                jx, jy = min(width - 1, ix + 1), min(height - 1, iy + 1)
                fx, fy = F(tx - F(ix)), F(ty - F(iy))
                indices = [iy * width + ix, iy * width + jx, jy * width + ix, jy * width + jx]
                channels = []
                for c in range(4):
                    byte = 2 - c if fmt == 87 and c < 3 else c
                    values = [F(F((int(textures[sample][i]) >> (8 * byte)) & 255) / F(255)) for i in indices]
                    if filters[sample]:
                        channels.append(values[0])
                    else:
                        upper = F(F(values[0] * F(F(1) - fx)) + F(values[1] * fx))
                        lower = F(F(values[2] * F(F(1) - fx)) + F(values[3] * fx))
                        channels.append(F(F(upper * F(F(1) - fy)) + F(lower * fy)))
                samples.append(channels)
            inverse = F(F(1) - F(enabled[1] * samples[1][3]))
            color = [F(min(1, max(0, F(F(inverse * F(enabled[0] * samples[0][c])) * alpha)))) for c in range(4)]
            for byte in range(4):
                channel = 2 - byte if target == 87 and byte < 3 else byte
                if not mask & (1 << channel):
                    continue
                value = F(color[channel] * F(255))
                if blend:
                    value = F(value + F(F((background >> (8 * byte)) & 255) * F(F(1) - color[3])))
                packed = int(np.rint(F(min(255, max(0, value)))))
                result = (result & ~(255 << (8 * byte))) | (packed << (8 * byte))
        struct.pack_into("<I", expected, 16 + n * 4, result)
    struct.pack_into("<8I", payload, 0, 0x4e524231, 1, count, vertices, cull, ccw, SPAN, 0)
    struct.pack_into("<4I", payload, 48, target, mask, blend, 0)
    for sample, resource in enumerate(resources):
        struct.pack_into("<3I", payload, 64 + sample * 12, *resource)
        struct.pack_into("<4f", payload, 96 + sample * 16, *clips[sample])
    struct.pack_into("<2I", payload, 88, *filters)
    struct.pack_into("<4I", expected, 0, 0x4e52424f, count, hits, 0)
    return bytes(payload), bytes(expected)
