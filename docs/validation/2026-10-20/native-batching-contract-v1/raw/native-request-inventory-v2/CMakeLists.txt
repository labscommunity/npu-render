cmake_minimum_required(VERSION 3.25)
project(nr_request_inventory LANGUAGES CXX)
find_package(OpenVINO 2026.1 REQUIRED COMPONENTS Runtime)
add_executable(nr_request_inventory inventory.cpp)
target_include_directories(nr_request_inventory PRIVATE "${NR_QUALIFIED_SOURCE}")
target_compile_features(nr_request_inventory PRIVATE cxx_std_17)
target_compile_options(nr_request_inventory PRIVATE /W4 /WX)
target_compile_definitions(nr_request_inventory PRIVATE NOMINMAX)
target_link_libraries(nr_request_inventory PRIVATE openvino::runtime bcrypt)
set_property(TARGET nr_request_inventory PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
