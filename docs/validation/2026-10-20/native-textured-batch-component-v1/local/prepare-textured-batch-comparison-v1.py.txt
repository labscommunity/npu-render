"""Prepare a serialized same-kernel texture batching comparison."""
from pathlib import Path
import ast,hashlib,json,subprocess
r=Path(__file__).parent
ast.parse((r/'compare-ptl-textured-batch-v1.py').read_text())
s=(r/'run-ptl-textured-batch-v1-qualification.ps1').read_text()
s=s.replace('ptl-textured-batch-v1-qualification','ptl-textured-batch-comparison-v1-qualification')
s=s.replace("@('ptl-worker-v31-qualification')","@('ptl-textured-batch-v1-qualification')")
s=s.replace('ptl-textured-batch-device-v1-source.json','ptl-textured-batch-comparison-v1-source.json')
a=s.index(" Invoke-Step 'target-compile'");b=s.index(' $report.passed=$true',a)
s=s[:a]+r''' Invoke-Step 'comparison' $python ('"'+$vm+'\compare-ptl-textured-batch-v1.py"') 360
 $q=Get-Content "$vm/ptl-textured-batch-comparison-v1/report.json" -Raw|ConvertFrom-Json
 if(!$q.completed -or !$q.passed -or @($q.child_result.cases).Count -ne 3){throw 'Comparison incomplete'}
 $report.exact_complete_images=42
'''+s[b:]
p=r/'run-ptl-textured-batch-comparison-v1.ps1';assert not p.exists();p.write_text(s)
s=(r/'start-textured-batch-v1.ps1').read_text().replace('NpuRender-TexturedBatch-v1','NpuRender-TexturedBatchComparison-v1').replace('run-ptl-textured-batch-v1-qualification.ps1','run-ptl-textured-batch-comparison-v1.ps1').replace('textured-batch-v1-task','textured-batch-comparison-v1-task')
p=r/'start-textured-batch-comparison-v1.ps1';assert not p.exists();p.write_text(s)
names=['compare-ptl-textured-batch-v1.py','ptl_textured_batch_fixtures_v1.py','run-ptl-textured-batch-comparison-v1.ps1','start-textured-batch-comparison-v1.ps1']
p=r/'ptl-textured-batch-comparison-v1-source.json';p.write_text(json.dumps({n:hashlib.sha256((r/n).read_bytes()).hexdigest() for n in names},indent=2)+'\n')
subprocess.run(['scp','-q',*[str(r/n) for n in names],str(p),'delta:AppData/Local/npu-render/vm/'],check=True,timeout=30)
print('Prepared 3 resolutions, 3 ABBA rounds each, separate/batched same kernel')
