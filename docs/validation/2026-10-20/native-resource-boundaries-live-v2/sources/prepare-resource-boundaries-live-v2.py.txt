pattern = re.compile("|".join(re.escape(x) for x in sorted(changes, key=len, reverse=True)))


def rename(s):
    return pattern.sub(lambda m: changes[m[0]], s)


files = [n for n in json.loads((r / "ptl-combined-live-v13-upload.json").read_text())
         if not n.endswith("-source.json")]
for name in files:
    s = rename((r / name).read_text())
    s = s.replace("native-desktop-admission-v70", "native-desktop-admission-v71")
    s = s.replace("native-registration-v49", "native-registration-v50")
    if name == "native_shared_app_v17.py":
        s = s.replace("from native_shared_correlation_v1 import", "from native_shared_correlation_v2 import")
        s = s.replace("native-shared-correlation-v1/report.json", "native-shared-correlation-v2/report.json")
        old = "   children=list((self.transaction/'bootstrap/provider-processes').glob('generation-*/worker.stdout.jsonl'));assert len(children)==1\n   child=complete_rows(children[0]);assert len(identities)==len(receipts)==len(acks)==8"

        assert s.count(old) == 1
        s = s.replace(old, new)
    if name == "run-native-live-v61.ps1":
        needle = " $ptlRows=@(Get-ChildItem "
        assert s.count(needle) == 1

 if(!$generationProof.completed -or !$generationProof.passed){throw 'Per-generation provider evidence incomplete'}

        for route in ("geometry", "program", "long", "complex", "rectangle", "resident_clip"):
            needle = "$report.ptl_" + route + "_imports -ne 1"
            assert s.count(needle) == 1
            s = s.replace(needle, "$report.ptl_" + route + "_imports -lt 1")
        start = s.index(" if($schedule.Count -ne 1)")
        end = s.index(" $report.active_phase='freshness-validation'", start)
        s = s[:start] + r''' if($schedule.Count -lt 1 -or $schedule.Count -ne $generationProof.balanced_schedules){throw 'Per-generation balanced geometry schedule not qualified'}
 $report.ptl_balanced_geometry_schedule=@($schedule)
''' + s[end:]

            s = s.replace("$q.ptl_" + route + "_imports -ne 1", "$q.ptl_" + route + "_imports -lt 1")
        needle = " if($q.ptl_quad_rectangle_executions"
        assert s.count(needle) == 1
