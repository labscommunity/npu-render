import argparse
import ast
import hashlib
import json
import shutil
import tarfile

parser = argparse.ArgumentParser()

args = parser.parse_args()
proof = json.loads(args.device_report.read_text(encoding="utf-8-sig"))
assert proof["completed"] and proof["passed"] and proof["child_result"]["unique_component_fixtures"] == 144
digest = proof["patched_sha256"]
assert len(digest) == 64 and all(c in "0123456789abcdef" for c in digest)
r = Path(__file__).parent
name = "ptl-native-worker-v23"
d = r / name
assert not d.exists()
shutil.copytree(r / "ptl-native-worker-v19", d)
s = (r / "ptl_resident_clip_v1.hpp").read_text().replace(
    "NR_RESIDENT_CLIP_QUALIFIED_DIGEST", ",".join("0x" + digest[n:n+2] for n in range(0, 64, 2))
)

    s = (r / "ptl-native-worker-v19" / ("ptl_program_triangle_draw." + extension)).read_text()
    s = s.replace("PtlProgramTriangleDrawExecutor", "PtlResidentClipTriangleDrawExecutor")
    s = s.replace("ptl_program_triangle_draw", "ptl_resident_clip_triangle_draw")
    s = s.replace("ptl_program.hpp", "ptl_resident_clip.hpp").replace("PtlProgramRaster", "PtlResidentClip")

a = s.index("    bool quad_enabled=false;")
s = s[:a] + r'''
    bool resident_clip_enabled=false;
    struct ResidentClipEntry {Header header;uint32_t batch;std::vector<uint32_t> vertex,pixel;raster::PackedResources layout;shader::TextureSamplers samplers;std::unique_ptr<raster::PtlResidentClipTriangleDrawExecutor> executor;};
    std::vector<ResidentClipEntry> resident_clips;
    std::optional<raster::UniformDrawExecutor::Result> try_resident_clip(const Request &request){
        if(!resident_clip_enabled||!program_shape(request)||(request.header.flags&0x2000)==0)return {};
        const auto &h=request.header;const auto capacity=vertex_capacity(h.vertices);
        const auto &resources=request.resources;const auto &samplers=request.samplers;
        if(!raster::PtlResidentClip::eligible(request.pixel,resources,samplers,capacity))return {};
        const auto clip=request.scissor.value_or(std::array<int32_t,4>{0,0,int32_t(h.width),int32_t(h.height)});
        require(scissor_valid(clip,h));const auto area=uint64_t(clip[2]-clip[0])*uint64_t(clip[3]-clip[1]);
        unsigned batch=raster::PtlResidentClip::maximum_batch(resources,capacity);
        while(batch>384&&area<=batch/2)batch/=2;
        raster::PtlResidentClipTriangleDrawExecutor *executor=nullptr;
        for(auto &entry:resident_clips){const auto &old=entry.header;
            if(entry.batch==batch&&old.vertices==capacity&&old.format==h.format&&(old.flags&~4u)==(h.flags&~4u)&&raster::resource_layout_equal(resources,entry.layout)&&raster::sampler_layout_equal(samplers,entry.samplers)&&entry.vertex==request.vertex.bytecode&&entry.pixel==request.pixel.bytecode){executor=entry.executor.get();break;}
        }
        if(!executor){
            if(resident_clips.size()==32)resident_clips.erase(resident_clips.begin());
            auto candidate=std::make_unique<raster::PtlResidentClipTriangleDrawExecutor>(request.vertex,request.pixel,capacity,h.width,h.height,h.format,resources,samplers,"NPU",blend_mode(h),uint8_t((h.flags>>8)&15),static_cast<raster::CullMode>((h.flags>>5)&3),(h.flags&128)!=0,batch);
            executor=candidate.get();auto key=h;key.vertices=capacity;resident_clips.push_back({key,batch,request.vertex.bytecode,request.pixel.bytecode,raster::resource_layout(resources),samplers,std::move(candidate)});
        }
        auto result=executor->execute(request.inputs,request.vertex_constants,request.pixel_constants,request.viewport,resources,request.background,request.scissor,request.blend_constant);
        std::printf("{\"ptl_resident_clip_draw\":true,\"accepted\":true,\"sequence\":%u,\"batch\":%u,\"provider_requests\":%llu,\"complete_texture_transport\":true,\"shader_exactly_pinned\":true,\"fallback_backend\":\"none\"}\n",h.sequence,batch,static_cast<unsigned long long>(result.provider_requests));std::fflush(stdout);
        return result;
    }
''' + s[a:]
s = s.replace("    Pipeline(){", r'''    Pipeline(){
        wchar_t *resident=nullptr;size_t resident_length=0;

        const std::unique_ptr<wchar_t,decltype(&std::free)> resident_owner(resident,&std::free);
        if(resident&&*resident){require(std::filesystem::path(resident).is_absolute());resident_clip_enabled=true;}
''')

    "CMakeLists.txt", "indexed_pipeline.hpp", "ptl_resident_clip.hpp",
    "ptl_resident_clip_triangle_draw.hpp", "ptl_resident_clip_triangle_draw.cpp",
}

    archive.add(d, arcname=name)

s = s.replace('|rectangle_tickets))', '|rectangle_tickets|{515, 518, 521, 524, 868, 402, 405, 533, 408, 536, 411, 743, 544, 744, 420, 745, 423, 746, 686, 431, 687, 688, 689, 690, 691, 692, 747, 316, 317, 318, 319, 320, 321, 322, 859, 862, 97, 225, 865, 100, 228, 741, 103, 231, 742, 106, 234, 620, 621, 622, 623, 624, 625, 626, 115, 243, 877, 118, 246, 880, 888, 126, 254}))')
s = s.replace(" env=os.environ.copy();", f""" resident=vm/'ptl-six-span-resident-clip-v3/patched.blob'
 assert obj(vm/'ptl-six-span-resident-clip-v3/report.json')['passed'] and sha(resident)=={digest!r}
 env=os.environ.copy();env['NR_RENDER_PTL_RESIDENT_CLIP_BLOB']=str(resident);""")
needle = " actual_program=routed_tickets(trial,'ptl_program_draw')"
assert s.count(needle) == 1
s = s.replace(needle, """ resident_tickets=routed_tickets(trial,'ptl_resident_clip_draw')
 assert resident_tickets and resident_tickets<=expected_geometry

 resident_events=[];resident_imports=[]

   if event.get('ptl_resident_clip_draw'):
    assert current_identity['pixel_sha256']=='b7f4e6d0abea95bb235098b7e357eacd906679d4626ba4abf4b3c5e3ca31ecf7'
    assert event['accepted'] and event['complete_texture_transport'] and event['shader_exactly_pinned'] and event['fallback_backend']=='none'
    assert 0<event['batch']<=6*16384 and event['provider_requests']>=2
    resident_events.append(event)

""" + needle)
s = s.replace(" report['passed']=True", """ if a.stage=='full':
  preceding=rows(vm/'ptl-native-worker-v19-full/replay/replays.jsonl');assert len(preceding)==972
  assert all((x['ticket'],x['pixels_sha256'],x['pixel_bytes'])==(y['ticket'],y['pixels_sha256'],y['pixel_bytes']) for x,y in zip(preceding,current,strict=True))
  report['all_worker_v19_outputs_preserved']=True
 report['passed']=True""")
