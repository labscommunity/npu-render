"""Separate the targeted quad draws from the complete recorded workload."""
from pathlib import Path
import hashlib
import json

b = Path("docs/validation/2026-10-20/native-quad-rectangle-v1")
p = b / "raw/quad-rectangle-comparison-v2/report.json"
q = json.loads(p.read_text(encoding="utf-8-sig"))
assert q["completed"] and q["passed"] and len(q["phases"]) == 4
tickets = {467, 587, 653, 719, 775}
phases = []
for phase in q["phases"]:
    selected = [x for x in phase["requests"] if x["original_ticket"] in tickets]
    assert len(selected) == 5
    phases.append(dict(phase=phase["phase"], worker_version=phase["worker_version"],
                       parent_ms=sum(x["parent_ms"] for x in selected),
                       child_ms=sum(x["child_ms"] for x in selected), requests=selected))
pairs = [x for x in q["paired_requests"] if x["original_ticket"] in tickets]
assert len(pairs) == 10
old = sum(x["old_parent_ms"] for x in pairs)
new = sum(x["new_parent_ms"] for x in pairs)
sources = [p]
calls = {}
for version, directory in [
    (19, Path("docs/validation/2026-10-16/native-response-packet-worker-v1/raw/ptl-native-worker-v19-full/replay")),
    (22, b / "raw/ptl-native-worker-v22-full-v2/replay"),
]:
    path = directory / "provider-processes/generation-1/worker.stdout.jsonl"
    sources.append(path)
    events = [json.loads(x) for x in path.read_text().splitlines() if x.startswith("{")]
    quads = [x for x in events if x.get("quad_batch_attempt") and x["accepted"]]
    assert len(quads) == 5 and sum(x["vertices"] for x in quads) // 6 == 19
    calls[str(version)] = sum(x["provider_requests"] for x in quads)
report = dict(completed=True, passed=True, new_device_execution=False,
              target_tickets=sorted(tickets), target_pairs=len(pairs),
              old_target_parent_ms=old, new_target_parent_ms=new,
              target_parent_reduction_percent=100 * (old - new) / old,
              slower_target_pairs=sum(x["new_parent_ms"] > x["old_parent_ms"] for x in pairs),
              complete_draw_provider_calls=calls, phases=phases,
              full_replay=q["summary"],
              source_sha256={str(path): hashlib.sha256(path.read_bytes()).hexdigest() for path in sources},
              scope="All five retained uses of the changed route; ten pairs from two runs per worker. Targeted speedup does not establish full-replay, live-desktop or Tahoma improvement.")
out = Path("results/render-driver/quad-rectangle-targets-v1")
out.mkdir(exist_ok=False)
(out / "report.json").write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps({k: v for k, v in report.items() if k not in ("phases", "source_sha256")}))
