"""Prepare short qualification then bounded thirty-minute native stability trials."""
from pathlib import Path
import hashlib
import json

r = Path('results/render-driver')


def changed(source, replacements):
    text = (r / source).read_text()
    for before, after in replacements:
        assert before in text, (source, before)
        text = text.replace(before, after)
    return text


def manifest(name, files):
    (r / name).write_text(json.dumps({file: hashlib.sha256((r / file).read_bytes()).hexdigest() for file in files}, indent=2) + '\n')


for name in ('run-native-registration', 'recover-native-registration'):
    s = changed(name + '-v22.ps1', [('v22', 'v23'), ('0.0.1316.0', '0.0.1317.0')])
    if name == 'run-native-registration':
        s = s.replace('native-driver-package-v16', 'native-driver-package-v17').replace('native-registration-stage-v16', 'native-registration-stage-v17').replace('native-routing-v34', 'native-routing-v41')
        s = s.replace("$principal=[Security.Principal.WindowsPrincipal]", "foreach($path in @('native-duration-components-v2','native-combined-v3')){$q=Get-Content \"$r\\vm\\$path\\report.json\" -Raw|ConvertFrom-Json;if(!$q.completed -or !$q.passed){throw 'Duration renderer prerequisites failed'}}\n$principal=[Security.Principal.WindowsPrincipal]", 1)
    (r / (name + '-v23.ps1')).write_text(s)
s = changed('qualify-native-registered-v22.py', [('0.0.1316.0', '0.0.1317.0')])
s = s.replace('report=dict(dll_sha256=', 'report=dict(worker_binary_sha256=hashlib.sha256(v.worker.read_bytes()).hexdigest(),dll_sha256=', 1)
(r / 'qualify-native-registered-v23.py').write_text(s)
manifest('native-registration-v23-source.json', ['run-native-registration-v23.ps1', 'recover-native-registration-v23.ps1', 'qualify-native-registered-v23.py', 'native_session_broker_v1.py'])
s = changed('observe_native_topology_v7.py', [('deadline=time.monotonic()+780', 'deadline=time.monotonic()+2340')])
(r / 'observe_native_topology_v8.py').write_text(s)

    needle = "foreach($name in @('T8-NpuRender-NativeTraceV1'"

    s = s.replace(needle, "$registeredWorker=Get-Content \"$r\\vm\\native-registration-v23\\qualification\\report.json\" -Raw|ConvertFrom-Json;if(!$registeredWorker.passed -or $registeredWorker.worker_binary_sha256 -ne (Get-FileHash $worker).Hash){throw 'Registered worker identity mismatch'}\n" + needle, 1)

        s = s.replace('$worker="', '$short=Get-Content "$r\\vm\\native-live-v28\\report.json" -Raw|ConvertFrom-Json;if(!$short.completed -or !$short.passed -or !$short.cleanup_passed){throw \'Short duration integration is not qualified\'}\n$worker="', 1)
