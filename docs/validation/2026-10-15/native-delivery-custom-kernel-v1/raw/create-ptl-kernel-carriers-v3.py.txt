"""Export and inspect unmodified Panther Lake graphs with exact add-one oracles."""
from pathlib import Path
from datetime import UTC,datetime
import hashlib,json,os,struct,time,traceback
import numpy as np
import openvino as ov
from openvino import opset13 as op
r=Path(os.environ['LOCALAPPDATA'])/'npu-render';out=r/'vm/ptl-kernel-carriers-v3';out.mkdir(exist_ok=False)
report=dict(started=datetime.now(UTC).isoformat(),completed=False,passed=False,custom_kernel_executed=False,desktop_changed=False,cases=[])
def inspect(blob):
 offset=blob.find(b'\x7fELF');assert 0<=offset<1024*1024
 data=blob[offset:];assert data[:6]==b'\x7fELF\x02\x01' and len(data)>=64
 h=struct.unpack_from('<16sHHIQQQIHHHHHH',data);shoff,shsize,count,names=h[6],h[11],h[12],h[13]
 assert shsize==64 and 0<count<65536 and names<count and shoff+count*shsize<=len(data)
 entries=[struct.unpack_from('<IIQQQQIIQQ',data,shoff+i*shsize) for i in range(count)]
 s=entries[names];assert s[4]+s[5]<=len(data);strings=data[s[4]:s[4]+s[5]];sections=[]
 # Intel ELF reader excludes NOBITS, CMX metadata and CMX workspace from file data.
 for s in entries:
  assert s[0]<len(strings) and (s[1] in (8,0x8aaaaaac,0x8aaaaaad) or s[4]+s[5]<=len(data))
  name=strings[s[0]:].split(b'\0',1)[0].decode('ascii')
  sections.append(dict(name=name,type=s[1],flags=s[2],offset=s[4],bytes=s[5],link=s[6],info=s[7],alignment=s[8],entry_bytes=s[9],sha256=hashlib.sha256(data[s[4]:s[4]+s[5]]).hexdigest() if s[1] not in (8,0x8aaaaaac,0x8aaaaaad) else None))
 return dict(elf_offset=offset,elf_class=64,type=h[1],machine=h[2],section_count=count,sections=sections)
try:
 core=ov.Core();assert 'NPU' in core.available_devices
 report['openvino_version']=ov.get_version();report['device_name']=str(core.get_property('NPU','FULL_DEVICE_NAME'))
 for dtype,kind in [(np.float16,ov.Type.f16),(np.float32,ov.Type.f32)]:
  name='add1-'+np.dtype(dtype).name;parameter=op.parameter([1,16],kind,name='input')
  model=ov.Model([op.add(parameter,op.constant(np.array(1,dtype=dtype)))],[parameter],name)
  start=time.monotonic();compiled=core.compile_model(model,'NPU',{'PERFORMANCE_HINT':'LATENCY'});compile_ms=(time.monotonic()-start)*1000
  raw_devices=compiled.get_property('EXECUTION_DEVICES');devices=[raw_devices] if isinstance(raw_devices,str) else list(raw_devices)
  report['observed_execution_devices']=devices;report['execution_property_type']=type(raw_devices).__name__
  assert len(devices)==1 and devices[0] in ('NPU','NPU.0'),'execution device must be the NPU'
  cases=[]
  for phase in range(8):
   data=(np.arange(16,dtype=np.float32).reshape(1,16)-8+phase/4).astype(dtype)
   output=compiled([data])[compiled.output(0)];expected=(data+np.array(1,dtype=dtype)).astype(dtype)
   assert output.dtype==expected.dtype and np.array_equal(output,expected)
   cases.append(dict(phase=phase,exact_components=16,input_sha256=hashlib.sha256(data.tobytes()).hexdigest(),output_sha256=hashlib.sha256(output.tobytes()).hexdigest()))
  exported=compiled.export_model();blob=exported.getvalue() if hasattr(exported,'getvalue') else bytes(exported)
  (out/(name+'.blob')).write_bytes(blob);metadata=inspect(blob)
  (out/(name+'.elf-metadata.json')).write_text(json.dumps(metadata,indent=2)+'\n')
  report['cases'].append(dict(name=name,execution_devices=devices,compile_ms=compile_ms,blob_bytes=len(blob),blob_sha256=hashlib.sha256(blob).hexdigest(),exact_components=128,inputs=cases,elf_class=metadata['elf_class'],section_count=metadata['section_count'],passed=True))
  (out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
 report['passed']=len(report['cases'])==2
except Exception as e:report.update(error=repr(e),traceback=traceback.format_exc())
finally:
 report.update(completed=True,finished=datetime.now(UTC).isoformat());(out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report));raise SystemExit(0 if report['passed'] else 1)
