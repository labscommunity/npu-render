"""Compare serial and overlapping requests using only previously admitted kernels."""
from pathlib import Path
from datetime import UTC,datetime
import argparse,hashlib,json,os,subprocess,sys,time,traceback
vm=Path(os.environ['LOCALAPPDATA'])/'npu-render/vm'
def write(p,v):p.write_text(json.dumps(v,indent=2)+'\n')
def obj(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def child(out):
 import openvino as ov
 import numpy as np

 report=dict(started=datetime.now(UTC).isoformat(),completed=False,passed=False,checks=[],health=[])

  assert ov.get_version()=='2026.4.1-22982-e213a147257-releases/2026/4';core=ov.Core()
  def imported(name):

  def health(phase):

   for i in range(4):

  health('before');shared=imported('patched');models=[imported('patched') for _ in range(6)];requests={'shared':[shared.create_infer_request() for _ in range(6)],'separate':[m.create_infer_request() for m in models]}
  indices=np.zeros(16384,np.int32)
  for group in range(6):
   sources=[];expected=[];input_hashes=[]

   for mode in (['serial','shared-overlap','separate-overlap'] if group%2==0 else ['separate-overlap','shared-overlap','serial']):
    rs=requests['separate' if mode=='separate-overlap' else 'shared'];actual=[];began=time.perf_counter_ns()
    if mode=='serial':
     for b,q in enumerate(rs):q.start_async({0:sources[b],1:indices});assert q.wait_for(10000);actual.append(q.get_output_tensor().data.copy())
    else:
     for b,q in enumerate(rs):q.start_async({0:sources[b],1:indices})

    elapsed=(time.perf_counter_ns()-began)/1e6;assert all(np.array_equal(a,e) for a,e in zip(actual,expected)),mode+' output mismatch'

   write(out/'child-report.json',report)
  del requests,models,shared,rs,q;health('after');report['passed']=True
 except Exception as error:report.update(error=repr(error),traceback=traceback.format_exc())
 finally:report.update(completed=True,finished=datetime.now(UTC).isoformat());write(out/'child-report.json',report)
