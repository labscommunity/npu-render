"""Verify a real NPU child is terminated when its owning renderer is killed."""
import ctypes as C
from ctypes import wintypes as W
from pathlib import Path
from datetime import UTC, datetime
import hashlib
import json
import os
import secrets
import subprocess
import time
import traceback

root = Path(os.environ['LOCALAPPDATA']) / 'npu-render'
out = root / 'vm/native-provider-owner-exit-v2'
out.mkdir(exist_ok=False)
worker_path = root / 'builds/native-routing-v34/Release/nr_registered_local_indexed_worker.exe'
client_path = root / 'builds/native-routing-v20/Release/nr_registered_local_indexed_client_probe.exe'
capture = root / 'vm/native-desktop-admission-v35/bootstrap/requests'
def obj(path):
    b = path.read_bytes()
    return json.loads(b.decode('utf-16') if b.startswith(b'\xff\xfe') else b.decode('utf-8-sig'))
def rows(path):
    return [json.loads(x) for x in path.read_text().splitlines() if x.startswith('{') and x.endswith('}')]
for binary in (worker_path, client_path):
    proof = obj(binary.parent.parent / 'registered-build-report.json')
    assert proof['clean_build'] and hashlib.sha256(binary.read_bytes()).hexdigest() == proof['binary_sha256'][binary.name].lower()
index = rows(capture / 'index.jsonl')
entry = next(x for x in index if x.get('request_recorded') and x['ticket'] == 0)
reference = next(x for x in index if x.get('provider_response_recorded') and x['ticket'] == 0)
packet = capture / entry['file']
assert hashlib.sha256(packet.read_bytes()).hexdigest() == entry['sha256'].lower()
env = os.environ.copy()
env['PATH'] = str(root / 'toolchains/openvino-2026.4.1/Lib/site-packages/openvino/libs') + os.pathsep + env.get('PATH', '')
for key in ('NR_RENDER_CAPTURE_DIRECTORY','NR_RENDER_AFFINE_BATCH','NR_RENDER_RECOVERY_TEST_KILL_TICKET','NR_RENDER_RECOVERY_TEST_EXCEPTION_TICKET','NR_RENDER_PROVIDER_TEST_INFER_EXCEPTION_TICKET'):
    env.pop(key, None)
children = out / 'provider-processes'
children.mkdir()
env.update(NR_RENDER_NPU_CACHE=str(root / 'vm/model-cache-2026-4-v209'), NR_RENDER_PIPELINE_CACHE='retained', NR_RENDER_SHARED_CORE='1', NR_RENDER_SHARED_QUEUE='1', NR_RENDER_MEMORY_CACHE='0', NR_RENDER_COMPACT_COVERAGE='0', NR_RENDER_OPAQUE_CROP='1', NR_RENDER_QUAD_BATCH='1', NR_RENDER_INFER_TRACE='1', NR_RENDER_COMPILE_TRACE='1', NR_RENDER_PROVIDER_RECOVERY='1', NR_RENDER_PROVIDER_RECOVERY_DIR=str(children), NR_RENDER_FAILURE_PACKET=str(out / 'failed-request.bin'))
k = C.WinDLL('kernel32', use_last_error=True)
k.OpenProcess.argtypes = [W.DWORD, W.BOOL, W.DWORD]
k.OpenProcess.restype = W.HANDLE
k.WaitForSingleObject.argtypes = [W.HANDLE, W.DWORD]
k.WaitForSingleObject.restype = W.DWORD
k.CloseHandle.argtypes = [W.HANDLE]
report = dict(started=datetime.now(UTC).isoformat(), completed=False, passed=False, desktop_changed=False, driver_installed=False, fault_injected=True, real_device_loss=False)
worker = None
child_handle = None
try:
    pipe = r'\\.\pipe\npu-render-indexed-' + secrets.token_hex(16)
    with (out / 'worker.stdout.jsonl').open('xb') as stdout, (out / 'worker.stderr.log').open('xb') as stderr:
        worker = subprocess.Popen([str(worker_path), pipe, '2', '120'], env=env, stdout=stdout, stderr=stderr, creationflags=subprocess.CREATE_NO_WINDOW)
        report['worker_pid'] = worker.pid
        deadline = time.perf_counter() + 15
        while not any(x.get('local_worker_ready') for x in rows(out / 'worker.stdout.jsonl')):
            assert worker.poll() is None and time.perf_counter() < deadline
            time.sleep(.02)
        result = subprocess.run([str(client_path), pipe, str(worker.pid), str(packet), str(out / 'reply.bin'), '90000', 'success'], env=env, capture_output=True, timeout=100)
        (out / 'client.stdout.jsonl').write_bytes(result.stdout)
        (out / 'client.stderr.log').write_bytes(result.stderr)
        assert result.returncode == 0 and not result.stderr
        reply = (out / 'reply.bin').read_bytes()
        assert hashlib.sha256(reply[128:]).hexdigest() == reference['pixels_sha256']
        (out / 'reply.bin').unlink()
        deadline = time.perf_counter() + 5
        while True:
            evidence = rows(out / 'worker.stdout.jsonl')
            if any(x.get('worker_success') for x in evidence):
                break
            assert time.perf_counter() < deadline
            time.sleep(.02)
        starts = [x for x in evidence if x.get('provider_process_started')]
        assert len(starts) == 1 and starts[0]['generation'] == 1
        report['child_pid'] = starts[0]['pid']
        child_handle = k.OpenProcess(0x100000 | 0x1000, False, report['child_pid'])
        assert child_handle and k.WaitForSingleObject(child_handle, 0) == 258
        began = time.perf_counter()
        worker.kill()
        worker.wait(timeout=5)
        assert k.WaitForSingleObject(child_handle, 5000) == 0, 'NPU child survived owner termination'
        report.update(passed=True, npu_request_acknowledged=True, exact_recorded_pixels=True, owned_child_terminated=True, cleanup_ms=(time.perf_counter()-began)*1000, parent_exit=worker.returncode)
except Exception as error:
    report.update(error=repr(error), traceback=traceback.format_exc())
finally:
    if worker is not None and worker.poll() is None:
        worker.kill()
        worker.wait(timeout=5)
    if child_handle:
        k.CloseHandle(child_handle)
    report.update(completed=True, finished=datetime.now(UTC).isoformat())
    (out / 'report.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report))
raise SystemExit(0 if report['passed'] else 1)
