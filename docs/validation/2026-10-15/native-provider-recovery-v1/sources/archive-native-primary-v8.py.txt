from pathlib import Path
from datetime import UTC, datetime
import difflib
import hashlib
import json
import re

r = Path('results/render-driver')
raw = r / 'primary-raw-v8'
out = Path('docs/validation/2026-10-15/native-provider-recovery-v1')
out.mkdir(parents=True, exist_ok=False)
def obj(path):
    b = path.read_bytes()
    return json.loads(b.decode('utf-16') if b.startswith(b'\xff\xfe') else b.decode('utf-8-sig'))
def rows(path):
    return [json.loads(x) for x in path.read_text(encoding='utf-8-sig').splitlines() if x.startswith('{') and x.endswith('}')]
manifest = obj(r / 'native-primary-v8-artifact-sha256.json')
for item in manifest:
    p = raw / item['source']
    b = p.read_bytes()
    assert len(b) == item['bytes'] and hashlib.sha256(b).hexdigest() == item['sha256'].lower()
    dest = out / 'raw' / item['source']
    if dest.suffix in ('.py', '.ps1'):
        dest = dest.with_suffix(dest.suffix + '.txt')
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(b)
    item['archive_path'] = dest.relative_to(out).as_posix()
(out / 'raw-manifest.json').write_text(json.dumps(manifest, indent=2) + '\n')
source_manifest = []
for version, base_version in [(31,28),(32,31),(33,32),(34,33)]:
    base = r / f'native-routing-v{base_version}'
    new = r / f'native-routing-v{version}'
    patch = []
    for name, sha in obj(raw / f'native-routing-v{version}-source.json').items():
        p = new / name
        assert hashlib.sha256(p.read_bytes()).hexdigest() == sha
        old = base / name
        if not old.exists() or old.read_bytes() != p.read_bytes():
            patch.extend(difflib.unified_diff(old.read_text().splitlines(keepends=True) if old.exists() else [],p.read_text().splitlines(keepends=True),fromfile='a/'+name if old.exists() else '/dev/null',tofile='b/'+name))
    dest = out / f'native-routing-v{version}-from-v{base_version}.patch'
    dest.write_text(''.join(patch))
    source_manifest.append(dict(path=dest.relative_to(out).as_posix(),sha256=hashlib.sha256(dest.read_bytes()).hexdigest(),kind='source-patch',base=f'native-routing-v{base_version}'))
for name, sha in obj(raw / 'infer-trace-contract-v1-source.json').items():
    p = r / 'infer-trace-contract-v1' / name
    assert hashlib.sha256(p.read_bytes()).hexdigest() == sha
    dest = out / 'sources/infer-trace-contract-v1' / (name+'.txt')
    dest.parent.mkdir(parents=True,exist_ok=True)
    dest.write_bytes(p.read_bytes())
    source_manifest.append(dict(path=dest.relative_to(out).as_posix(),sha256=sha,kind='contract-source'))
for name in ['prepare-native-routing-v31.py','prepare-native-routing-v32.py','prepare-native-routing-v33.py','prepare-native-routing-v34.py','provider-process-v1.hpp','collect-native-primary-v8.py','archive-native-primary-v8.py']:
    p=r/name;dest=out/'sources'/(name+'.txt');dest.write_bytes(p.read_bytes());source_manifest.append(dict(path=dest.relative_to(out).as_posix(),sha256=hashlib.sha256(p.read_bytes()).hexdigest(),kind='local-experiment-source'))
(out/'source-manifest.json').write_text(json.dumps(source_manifest,indent=2)+'\n')
failure=obj(raw/'native-ordered-replay-v31-live24-traced/report.json')
soak=obj(raw/'native-ordered-replay-v32-stage-soak/report.json')
assert failure['completed'] and not failure['passed'] and failure['successful_replays']==820
assert 'ZE_RESULT_ERROR_DEVICE_LOST' in (raw/'native-ordered-replay-v31-live24-traced/worker.stderr.log').read_text()
assert soak['passed'] and soak['successful_replays']==3291 and soak['original_pixel_hashes_matched']==3288 and soak['unreferenced_successes']==3 and soak['delivery_acknowledgments']==3291
qualified=[obj(raw/f'native-provider-process-v{v}/report.json') for v in [1,2]]
for q in qualified:
    assert q['passed'] and len(q['cases'])==3 and all(c['pixel_hashes_matched']==c['delivery_acknowledgments']==5 for c in q['cases'])
    assert q['owner_exit']['passed'] and q['owner_exit']['owned_child_terminated']
trace=obj(raw/'native-ordered-replay-v31-live24-traced-control/trace-summary-v3/report.json')
summary=(raw/'native-ordered-replay-v31-live24-traced-control/trace-summary-v4-loss/summary.txt').read_bytes()
summary=summary.decode('utf-16') if summary.startswith(b'\xff\xfe') else summary.decode('utf-8-sig')
lost=int(re.search(r'Total Events\s+Lost\s+(\d+)',summary)[1]);assert lost==628
compiler=rows(raw/'native-ordered-replay-v33-compile-identities/worker.stdout.jsonl')
draw=next(x for x in compiler if x.get('execution_device')=='NPU' and x.get('sequence')==1054)
