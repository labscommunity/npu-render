"""Narrow full-source FP16 Gather carrier admission for the observed PTL graphs."""
import hashlib,struct
from ptl_elf_v1 import Elf,check

def inspect(blob,elements):
 check(elements in (256,4096),'qualified source extent')
 elf=Elf(blob);view=elf.describe(3)
 check(not any('dpu' in s['name'].lower() for s in elf.sections),'no DPU carrier')
 check(len(view['ranges'])==len(view['invocations'])==1,'single invocation carrier')
 check(len([s for s in elf.sections if s['name'].startswith('task.shave.')])==2,'no extra ACT records')
 code=elf.by_name['shave.text'];params=elf.by_name['shave.params'];data=elf.by_name['shave.data'];rs=view['ranges'][0];call=view['invocations'][0]
 check(code['type']==params['type']==data['type']==1 and not elf.relocations(code),'code and parameter section types')
 check(rs['name']=='task.shave.range.0.0' and rs['kind']==rs['base']==0 and rs['entry']==0x1d000000 and 0<rs['code_bytes']<=code['size'],'range entry and extent')
 check(rs['relocations']=={16:dict(kind=0,section='shave.text',addend=0)},'code selection')
 check(call['name']=='task.shave.invocation.0.0' and call['tile']==0,'single tile identity')
 check(call['relocations']=={8:dict(kind=0,section='shave.params',addend=0),16:dict(kind=0,section='shave.data',addend=0)},'invocation parameter references')
 prs=elf.relocations(params);check(set(prs)=={16,20,56,60,96,100},'parameter relocations')
 expected=[(2,[4,elements],[16,64],2,0x21),(1,[64],[32],9,1),(2,[4,64],[16,64],2,0x21)]
 spans=[]
 for i,(tensor,expected_fields) in enumerate(zip(call['tensors'],expected)):
  rank,dims,strides,dtype,order=expected_fields
  check((tensor['rank'],tensor['dims'],tensor['strides_bits'],tensor['dtype'],tensor['order'])==expected_fields,'dense interleaved tensor contract')
  check(tensor['location']==2 and tensor['address_relocation'] is None and 0x40200000<=tensor['address']<0x40400000,'CMX tensor pointer')
  width=elements*8 if i==0 else (256 if i==1 else 512)
  check(tensor['address']+width<=0x40400000,'CMX tensor extent');spans.append((tensor['address'],tensor['address']+width))
  for field,size in [(16,rank*4),(20,rank*8)]:
   rel=prs[i*40+field];check(rel['kind']==4 and rel['section']=='shave.params' and 120<=rel['addend'] and rel['addend']+size<=params['size'],'descriptor shape arrays')
 check(spans[0][0]==0x40218000 and spans[0][1]==spans[1][0] and spans[1][1]==spans[2][0],'disjoint full tensor allocation')
 return dict(accepted=True,blob_sha256=hashlib.sha256(blob).hexdigest(),source_elements=elements,source_components=elements*4,index_elements=64,output_components=256,code_offset=code['offset'],code_bytes=rs['code_bytes'],invocations=view['invocations'],full_source_available_to_single_invocation=True,custom_execution_qualified=False)

def patch(blob,text,elements):
 contract=inspect(blob,elements);check(0<len(text)<=contract['code_bytes'] and len(text)%16==0,'replacement extent');start=contract['code_offset'];size=contract['code_bytes'];modified=blob[:start]+text+bytes(size-len(text))+blob[start+size:]
 check(len(modified)==len(blob) and modified[:start]==blob[:start] and modified[start+size:]==blob[start+size:],'non-code bytes preserved');inspect(modified,elements);return modified,contract
