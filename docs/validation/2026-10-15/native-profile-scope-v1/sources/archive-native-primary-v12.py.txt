"""Retain profiled costs, a failed uniform candidate, and scoped-upload timing."""

from datetime import UTC,datetime
import collections,difflib,hashlib,json,math
r=Path('results/render-driver');raw=r/'primary-raw-v12';out=Path('docs/validation/2026-10-15/native-profile-scope-v1')
def obj(p):
 b=p.read_bytes();return json.loads(b.decode('utf-16') if b.startswith(b'\xff\xfe') else b.decode('utf-8-sig'))
def rows(p):return [json.loads(s) for s in p.read_text(encoding='utf-8-sig').splitlines() if s.startswith('{') and s.endswith('}')]
manifest=obj(r/'native-primary-v12-artifact-sha256.json')
for version in [39,40,42,43]:
 build=obj(raw/f'builds/native-routing-v{version}/registered-build-report.json')
 assert build['clean_build'] and build['admission_contract_passed']
 assert build['source_sha256']==obj(raw/f'native-routing-v{version}-source.json')
profile=obj(raw/'native-stage-profile-v1-recorded/report.json')
assert profile['passed'] and profile['original_pixel_hashes_matched']==profile['delivery_acknowledgments']==16
child=rows(raw/'native-stage-profile-v1-recorded/provider-processes/generation-1/worker.stdout.jsonl')
draws=[d for d in child if d.get('passed') and 'draw_ms' in d]
profiles=[d for d in child if d.get('provider_draw_profile')];assert len(profiles)==len(draws)==16
sites=collections.defaultdict(lambda:dict(calls=0,inference_ms=0.))
for p,d in zip(profiles,draws):
 assert p['sequence']==d['sequence'] and p['unclassified_inferences']==0
 assert sum(s['calls'] for s in p['sites'])==d['provider_requests']
 for s in p['sites']:
  assert s['failed']==0;sites[s['site']]['calls']+=s['calls'];sites[s['site']]['inference_ms']+=s['inference_ms']
packed=obj(raw/'packed-texture-gather-v1/report.json');assert packed['passed'] and packed['completed']
assert all(m['execution_devices']==['NPU'] and m['three_index_patterns_exact'] for m in packed['models'])
uniform1=obj(raw/'native-uniform-lanes-v1-evidence/report.json');uniform2=obj(raw/'native-uniform-lanes-v2-evidence/report.json')
assert uniform1['completed'] and not uniform1['passed'] and uniform1['active_phase']=='contract-CPU-0'
assert uniform2['completed'] and not uniform2['passed'] and uniform2['active_phase']=='warm-B'
assert len(uniform2['phases'])==5 and all(p['passed'] for p in uniform2['phases'])
assert all(p['exact_components']==9442300 for p in uniform2['phases'][:4])
failure=obj(raw/'native-uniform-lanes-v2-warm-B/report.json');assert failure['completed'] and not failure['passed'] and 'Pixels differ at ticket 140' in failure['error']
scope=obj(raw/'native-texture-scope-v1-evidence/report.json');assert scope['completed'] and scope['passed']
assert [p['phase'] for p in scope['phases']]==['contract-CPU','contract-NPU','warm-A','warm-B','A1','B1','B2','A2']
for phase in scope['phases'][2:]:
 trial=raw/('native-texture-scope-v1-'+phase['phase']);q=obj(trial/'report.json')
 assert q['passed'] and q['successful_replays']==q['original_pixel_hashes_matched']==q['delivery_acknowledgments']==16
 parent=rows(trial/'worker.stdout.jsonl');assert sum(bool(p.get('provider_process_started')) for p in parent)==1
 assert not any(p.get('provider_process_failure') for p in parent)
 ds=[p for p in rows(trial/'provider-processes/generation-1/worker.stdout.jsonl') if p.get('passed') and 'compile_and_draw_ms' in p];assert len(ds)==16
 for k,v in phase['child_totals'].items():assert math.isclose(sum(p[k] for p in ds),v,abs_tol=.00001)
 assert phase['before_power']['ac_online']==phase['after_power']['ac_online']==1
 if not phase['phase'].startswith('warm-'):assert phase['child_totals']['cache_hits']==phase['child_totals']['cache_requests']
for mode in 'AB':
 mean=sum(p['child_totals']['compile_and_draw_ms'] for p in scope['phases'] if p['phase'] in (mode+'1',mode+'2'))/2
 assert math.isclose(mean,scope['mean_compile_draw_ms'][mode])
out.mkdir(parents=True,exist_ok=False)
for item in manifest:
 data=(raw/item['source']).read_bytes();assert len(data)==item['bytes'] and hashlib.sha256(data).hexdigest()==item['sha256'].lower()
 destination=out/'raw'/item['source']
 if destination.suffix in ('.py','.ps1'):destination=destination.with_suffix(destination.suffix+'.txt')
 destination.parent.mkdir(parents=True,exist_ok=True);destination.write_bytes(data);item['archive_path']=destination.relative_to(out).as_posix()
(out/'raw-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
sources=[]
for base,new in [(37,39),(39,40),(39,42),(42,43)]:
 patch=[]
 for name,checksum in obj(raw/f'native-routing-v{new}-source.json').items():
  old=r/f'native-routing-v{base}'/name;path=r/f'native-routing-v{new}'/name
  assert hashlib.sha256(path.read_bytes()).hexdigest()==checksum

 destination=out/f'native-routing-v{new}-from-v{base}.patch';destination.write_text(''.join(patch))
 sources.append(dict(path=destination.name,sha256=hashlib.sha256(destination.read_bytes()).hexdigest(),base=f'native-routing-v{base}'))
for name in ['prepare-native-routing-v39.py','prepare-native-routing-v40.py','prepare-native-routing-v42.py','collect-native-primary-v12.py','archive-native-primary-v12.py']:
 destination=out/'sources'/(name+'.txt');destination.parent.mkdir(parents=True,exist_ok=True);destination.write_bytes((r/name).read_bytes());sources.append(dict(path=destination.relative_to(out).as_posix(),sha256=hashlib.sha256(destination.read_bytes()).hexdigest()))
(out/'source-manifest.json').write_text(json.dumps(sources,indent=2)+'\n')
